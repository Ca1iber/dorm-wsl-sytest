phHookActionDone

Subtopics

  1. Parameters:
  2. User-Specific Code:

This hook function is called (if activated, see Activating a Hook Function) on every PROB_HND_CALL from the application model, after the driver has performed all necessary actions for this call.

The simple example functionality of the hook function template demonstrates how to use the function phStateGetSiteInfo.

Appendix B: Man Pages for Driver Functions describes how to get information about this function, and about other functions which can be called from within a hook function.

phHookError_t phHookActionDone(
    phFrameAction_t call/* the identification of an incoming
                         call from the application model */,
    phEventResult_t *result/* the result of this hook call */,
    char *parm_list_input/* the parameters passed over from
                          the application model */,
    int parmcount/* the parameter count */,
    char *comment_out/* the comment to be send back */
                    /* to the application model , as */
                    /* already set by the driver */
    int *comlen/* the length of this comment */,
    phTcomUserReturn_t *stReturn  /* SmarTest return value, */
                                  as  already set by the driver */
)
{
    /* some local variables */

#ifdef ACTION_DONE_IMPLEMENTED
    phStateSiteUsage_t *sitePopulation;
    int entries;
    int i;
    static long deviceCount = 0L;
#endif

    /* do checks and trace the call first */
    CheckInitialized(&hook);
    phLogHookMessage(hook.myLogger, PHLOG_TYPE_TRACE, 
                     "phHookActionDone(0x%08lx,\"%s\",%d,\"%s\",%d,P%p)",
                     (long) call,
                     parm_list_input ? parm_list_input : "(NULL)", parmcount, 
                     comment_out ? comment_out : "(NULL)", comlen, 
                     stReturn);

#ifdef ACTION_DONE_IMPLEMENTED

    /* Here is a sample implementation which prints out a
    message after each performed ph_device_start call and 
     ph_device_start call from the application model, and
    calculates the device count */

    switch (call)
    {
      case PHFRAME_ACT_DEV_START:

      /* get the handler's site population status and count all devices */

      if (phStateGetSiteInfo(hook.myState, &sitePopulation,
          &entries) == PHSTATE_ERR_OK)
      {
        for (i=0; i<entries; i++)
        if (sitePopulation[i] == PHSTATE_SITE_POPULATED)
        deviceCount++;
      }

      phLogHookMessage(hook.myLogger, 
                       PHLOG_TYPE_MESSAGE_0,
                       "end of ph_device_start recognized by hook function,
    total: %ld", deviceCount);
     break;

     case PHFRAME_ACT_DEV_DONE:
       phLogHookMessage(hook.myLogger, PHLOG_TYPE_MESSAGE_0,
                       "end of ph_device_done recognized by hook function");
     break;

     default:
       /* nothing to do */
     break;
    }

    return PHHOOK_ERR_OK;

#else

    /* hook was not activated */
    return PHHOOK_ERR_NA;

#endif
}
Note For all type definitions, see the header files corresponding to the respective type. The header files are located in the directory ... /GenericHandler/include/ and in ... /GenericProber/include For example, for the type phTcomUserReturn_t see the header file phTcom.h.
