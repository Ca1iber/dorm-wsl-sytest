BfdGenerator

Subtopics

  1. BfdGenerator()
  2. BfdGenerator.addFailmap()

The BfdGenerator class lets you create bit fail maps from data that have been generated by a custom test method or other C++ progam, making the Bit Fail Map Viewer available for examining the data.

Note The BfdGenerator class is only available when the MTP package is installed.

Using BfdGenerator

To be able to use BfdGenerator, you must include the bfd_gen.h header file in your code and link to the libbfdapi.so shared library when building the test method or program.

These files are installed at:

  • Header file: /opt/93000mts/mts/result_server/include/bfd_gen.h
  • Shared library: /opt/93000mts/mts/result_server/sh_lib/libbfdapi.so

Examples

Example 1

This example generates one BDF file in ASCII format, populated with a few errors:

int testSmallFailuresAsciiBfd()
{
  string bfdFileName = "/tmp/testSmallFailuresAsciiBfd.bfd";
  string fileComments = "testing bfd generation APIs";
  long xBits = 11, yBits= 9, zBits = 0, dBits = 12;
  char *encodingSequence = "dzyx";
  enum EncodingMode mode = ASCII_ENCODING;

  BfdGenerator bfdGen(bfdFileName, fileComments, xBits, yBits, zBits, dBits, encodingSequence, mode);

  string failmap0 = "smallFailures";
  const int size = 2;
  long long nFailAccesses0 = size;
  long long nFailCells0 = size;

  Address addrList0[size];
  addrList0[0].x = 19;
  addrList0[0].y = 26;
  addrList0[0].z = 0;
  addrList0[0].d = 9;
  addrList0[0].expD = 0;

  addrList0[1].x = 9;
  addrList0[1].y = 126;
  addrList0[1].z = 0;
  addrList0[1].d = 2;
  addrList0[1].expD = 1;

  bfdGen.addFailmap(failmap0, nFailAccesses0, nFailCells0, addrList0, size);

  return 0;
}

Example 2

This example generates one BDF file in binary format that contains a lot of failures:

int testLargeFailuresBinaryBfd()
{
  string bfdFileName = "/tmp/testLargeFailuresBinaryBfd.bfd";
  string fileComments = "testing bfd generation APIs";
  long xBits = 11, yBits= 9, zBits = 0, dBits = 12;
  char *encodingSequence = "dzyx";

  BfdGenerator bfdGen(bfdFileName, fileComments, xBits, yBits, zBits, dBits, encodingSequence);

  string failmap0 = "largeFailures";
  const int size = 10000;
  long long nFailAccesses0 = size;
  long long nFailCells0 = size;

  Address addrList0[size];
  long xSize = (1 << xBits), ySize = (1 << yBits), zSize = (1 << zBits);

  for(int i=0; i < size; i++)
  {
    addrList0[i].x = i % xSize;
    addrList0[i].y = i % ySize;
    addrList0[i].z = i % zSize;
    addrList0[i].d = i % dBits;
    addrList0[i].expD = i % 2;
  }

  bfdGen.addFailmap(failmap0, nFailAccesses0, nFailCells0, addrList0, size);

  return 0;
}

Example 3

This example generates one BFD file in ASCII format that contains multiple bit fail maps:

int testSeveralFailmapInOneBfd()
{
  string bfdFileName = "/tmp/testSeveralFailmapInOneBfd.bfd";
  string fileComments = "testing bfd generation APIs";
  long xBits = 11, yBits= 9, zBits = 0, dBits = 12;
  char *encodingSequence = "dzyx";
  enum EncodingMode mode = ASCII_ENCODING;

  BfdGenerator bfdGen(bfdFileName, fileComments, xBits, yBits, zBits, dBits, encodingSequence, mode);

  long int nFailmap = 5;

  for (int n =0; n < nFailmap; n++)
  {
    string failmap = "failmap-";
    failmap += char('0'+n);
    int size = (n+1)*100/(n+2);
    long long nFailAccesses = size;
    long long nFailCells = size;

    Address *addrList = new Address[size];
    long xSize = (1 << xBits), ySize = (1 << yBits), zSize = (1 << zBits);
    for(int i=0; i < size; i++)
    {
      addrList[i].x = (i+n) % (xSize);
      addrList[i].y = (i+n) % (ySize);
      addrList[i].z = (i+n) % (zSize);
      addrList[i].d = (i+n) % (dBits);
      addrList[i].expD = (i+n) % 2;
    }

    bfdGen.addFailmap(failmap, nFailAccesses, nFailCells, addrList, size);

    delete[] addrList;
  }

  return 0;
}
