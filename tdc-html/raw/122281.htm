How to implement a concurrent test method

To implement a concurrent test method, follow the structure provided in the template file.

Before you begin

A concurrent test method class was created.

Procedure

  1. In the test method class file, define your own test method parameters and limits in the addParameters() function with the API addParameter and addLimit.
  2. In the init() function:
    • configure the execution mode with CCTsetExecMode API.
    • turn on the support of SmartCalc with the CCTenableSMC aPI if needed.
    • register digitizer pins with CCTregisterDGT and register RF Tx pins with CCTregisterTx if needed
    • configure the result area for digital capture hidden upload with the CCTconfigResultArea aPI if needed
    CCTsetExecMode(EXM_EXECUTE_TEST);
CCTregisterDGT(dac_out);
  3. In the preStart() function, insert initialization code for the test. For example you can enable HW triggers here.
  4. In the postStart() function, you can invoke the static DC tests that do not interfere with other sequencers. 
    SPMU_TASK task;
ON_FIRST_INVOCATION_BEGIN();
  task.pin("pin1").settling(2 ms).relay("NTBBM")
      .vForce(250 mV).clamp(1 mA).limits("limits1");
  task.preAction("FTST").execMode("PVAL").execute();
ON_FIRST_INVOCATION_END();
DOUBLE result=task.getValue("pin1");

    If you are doing spec-search like test on the independent sequencer, put the CCT_ContinueCurrentTest in this function.

  5. In the process() function, insert the result upload, calculation and datalog code. Use SmartCalc API here when you have enabled SmartCalc support. 
    dgtz_waveform = DGT(dac_out).getWaveform(CCTgetUploadId(dac_out));
DSP_SPECTRUM(dgtz_waveform, spect_waveform, DB, ref_level_DAC, RECT, 0);

    For the tests running on the independent sequencer, you can retrieve the results while the synchronous sequencer is running. (For details, refer to CCF-ISR mode.)

  6. In the cleanup() function, insert the code to clean up the environment, like disabling/disconnecting current instrument (AWG, DGT). 
    DGT(dac_out).disable();
Routing.pin(dac_out).disconnect();

Results

A concurrent test method is created with the concurrent test framework.

Example

The following example shows how to implement the concurrent test method for DAC distortion test. 

#include "CCTestMethod.hpp"
#include "TMLimits.h"

using namespace cct;
using V93kLimits::tmLimits;

class DAC_Distortion : public cct::CCTestMethod
{
  protected:
    string   dac_out;
    int      num_cycle_DAC;
    THDtype  thd_result;
    ARRAY_D  dgtz_waveform;
    ARRAY_D  spect_waveform;

  protected:
    virtual void addParameters()
    {
      addParameter("dac_out", "string", &dac_out)
        .setOptions("DAC_WDE:DAC.VOUT.MCA:DAC.VOUT.MCB")
        .setComment("DAC sequence 2");
      addParameter("num_cycle_DAC", "int", &num_cycle_DAC)
        .setDefault("25");
    }
    virtual void init()
    {
      CCTsetExecMode(EXM_EXECUTE_TEST);
      CCTregisterDGT(dac_out);
    }

    virtual void preStart()
    {
      // Connect the DAC pin and enable the digitizer
      Analog.primary(CCTgetAnalogRFSetName());
      Routing.pin(dac_out).connect(TM::SINGLE);
      Analog.DGT(dac_out).enable();
    }
    virtual void postStart()
    {
      // No postStart actions.
    }
    virtual void process()
    {
      INT Offline_flag;
      INT    num_harmonics_DAC = 5;
      DOUBLE ref_level_DAC = 0.0;

      GET_SYSTEM_FLAG("offline", &Offline_flag); 
      // Get Digitized Waveform Data
      dgtz_waveform = DGT(dac_out).getWaveform(CCTgetUploadId(dac_out));
      // Generate Offline Data
      if(Offline_flag) 
      {
        dgtz_waveform.resize(4096);
        for(int i=0;i<4096;i++) 
        {
          DOUBLE var=2*M_PI*i/4096*13;
          dgtz_waveform[i]=4095*sin(var);
        }
      }
      INT num_pts = dgtz_waveform.size();
      spect_waveform.resize(num_pts/2);
      DSP_SPECTRUM(dgtz_waveform, spect_waveform, DB, ref_level_DAC, RECT, 0);
      // Result Evaluation
      DSP_THD(dgtz_waveform, &thd_result, num_cycle_DAC, num_harmonics_DAC, DB, 0);
      // Pass/Fail Evaluation & Datalog
      string tsName = CCTgetTestsuiteName(); // Get the original testsutie name
      cout << "INFO: testsuite name: " << tsName << endl;
      cout << thd_result.snr << " -> DAC_SNR" << endl;
      cout << thd_result.thd << " -> DAC_THD" << endl;
      // Get the software bin name and hardware bin number from tm limit file
      // NOTE: the test limit file (.csv) should be specified in testflow setup
      bool pass_1;
      bool pass_2;
      string softBinNum_1 = tmLimits.getLimit(tsName, "DAC_SNR").BinsNumString; 
      int hardBinNum_1 = tmLimits.getLimit(tsName, "DAC_SNR").BinhNum;
      string softBinNum_2 = tmLimits.getLimit(tsName, "DAC_THD").BinsNumString; 
      int hardBinNum_2 = tmLimits.getLimit(tsName, "DAC_THD").BinhNum;
      string softBinNum_3 = tmLimits.getLimit(tsName, "DAC_SNR_THD").BinsNumString; 
      int hardBinNum_3 = tmLimits.getLimit(tsName, "DAC_SNR_THD").BinhNum;
      // Test against limit and datalog
      pass_1 = TestSet.cont(TM::CONTINUE)
                      .judgeAndLog_ParametricTest(dac_out,
                         "DAC_SNR",
                         tmLimits.getLimit(tsName, "DAC_SNR").TEST_API_LIMIT, 
                         thd_result.snr);
      pass_2 = TestSet.cont(TM::CONTINUE)
                      .judgeAndLog_ParametricTest(dac_out,
                        "DAC_THD",
                        tmLimits.getLimit(tsName, "DAC_THD").TEST_API_LIMIT, 
                        thd_result.thd);

     // Multibin is used for binning.
     // Set multibin according to different result branch
     if(!pass_1&&pass_2) SET_MULTIBIN(softBinNum_1, hardBinNum_1);
     if(pass_1&&!pass_2) SET_MULTIBIN(softBinNum_2, hardBinNum_2);
     if(!pass_1&&!pass_2) SET_MULTIBIN(softBinNum_3, hardBinNum_3);
    }

    virtual void cleanup()
    {
      DGT(dac_out).disable();
      Routing.pin(dac_out).disconnect();
    }

    virtual void postParameterChange(const std::string& parameterIdentifier)
    {
      //Add your code
    }
};
REGISTER_TESTMETHOD("DAC_Distortion", DAC_Distortion);
