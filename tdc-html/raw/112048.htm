Two complete examples of using HVM API

Two examples are provided here. One is for test execution controlling and one is for production status monitoring.

Example of downloading and executing a test program

#include <xoc/hvmapi/hvmapi.hpp>
#include <iostream>
#include <cstdio>

using namespace std;

static std::string sSession;

// handlerSession() changes device for the first SmarTest session 
// and report the session information for other conditions.
void handlerSession (const std::string& type,
                     const xoc::hvmapi::ParamList& params,
                     unsigned long sec,
                     long usec,
                     const std::string& session)
{
  using namespace xoc::hvmapi;
  Status errObj;
  
  static int nCountOfSessionReadyEvent = 0;

  if (sSession == session) 
  {
    if (params["SESSION_STATE"] == "SMARTEST_SESSION_READY") 
    {
      nCountOfSessionReadyEvent++;
    
      cout << "received SMARTEST_SESSION_READY" << endl;
      
      cout << "number of SMARTEST_SESSION_READY is " 
           <<  nCountOfSessionReadyEvent << endl;


      if ( nCountOfSessionReadyEvent == 1) 
      {
  
        xoc::hvmapi::ParamList cmdParam;

        cmdParam["DEVICE_PATH"] = "/home/demo/devices/";
        cmdParam["DEVICE_NAME"] = "hvmapi_device.src";
 

        cout << "FA: ChangeDevice started: " << endl;
        errObj = xoc::hvmapi::command ("ChangeDevice",cmdParam,session);
        if ( errObj.isSuccess() ) 
        {
          cout << "FA: ChangeDevice executed: " << endl;
        } 
        else 
        {
          cout << "FA ChangeDevice not executed" << endl;
       cout << "error code is " << errObj.code() << endl;
        }
      } 
      else 
      {
        cout << "received second SESSION_READY, do not change the device" 
             << endl;        
      }

      cout << " any key to continue " << endl;
      
      getchar();
                  
    }
  } 
  else 
  {
    string s = params["SESSION_STATE"];
    
    cout << " received other session event: " << s <<endl;
  }
  
  return;
}

// handlerDeviceChangeComplete() downloads test program ft1
// for a SmarTest session
void handlerDeviceChangeComplete (const std::string& type,
                                  const xoc::hvmapi::ParamList& params,
                                  unsigned long sec,
                                  long usec,
                                  const std::string& session)
{
  if (sSession == session) 
  {

    xoc::hvmapi::ParamList cmdParam;
    cmdParam.setValue ("TESTPROGRAM_NAME","ft1");

    cout << " received change device completed events " << endl;
    
    cout << " any key to download the testprogram" << endl;
    
    getchar();
    
    cout << "FA: DownloadTestprogram started: " << endl;

    xoc::hvmapi::Status errObj = xoc::hvmapi::command("DownloadTestprogram",cmdParam,session);
    
    cout << "FA: DownloadTestprogram executed: " << endl;


    if ( errObj.isSuccess() ) 
    {
      cout << "FA: DownloadTestprogram executed: " << endl;
    } 
    else 
    {
      cout << "FA DownloadTestprogram not executed" << endl;
    }
    
  }
  return;
}

// handlerTPLoadComplete() executes the test program 
//after the test program is completely downloaded.
void handlerTPLoadComplete (const std::string& type,
                            const xoc::hvmapi::ParamList& params,
                            unsigned long sec,
                            long usec,
                            const std::string& session)
{
  if (sSession == session) 
  {
    cout << " receive download testprogram completed event" << endl;
    cout << " any key execute the testprogram " << endl;
    getchar();

    xoc::hvmapi::ParamList cmdParam;
    xoc::hvmapi::Status errObj = xoc::hvmapi::command
                                ("ExecTestprogram",cmdParam,session);
    
    cout << "FA: ExecTestprogram executed: " << endl;


    if ( errObj.isSuccess() ) 
    {
        cout << "FA: ExecTestprogram successfully executed: " << endl;
    } 
    else 
    {
        cout << "FA ExecTestprogram not executed" << endl;
    }
          

  }
  return;
}

//handlerTPComplete() exit the HVM internal thread 
//when the test program execution is finished.
void handlerTPComplete (const std::string& type,
                        const xoc::hvmapi::ParamList& params,
                        unsigned long sec,
                        long usec,
                        const std::string& session)
{
  if (sSession == session) 
  {
    xoc::hvmapi::eventLoopExit ();
    cout << "FA: eventLoopExit executed " << endl;
  }
  return;
}

// The main program registers the event handler defined above, 
// start an offline SmarTest session, 
// enter the event loop and proceed based on the event occurs.

int main(int argc, char **argv) 
{
  xoc::hvmapi::Status errObj;
  
 cout << "FA: Register handlers" << endl;
  if (!xoc::hvmapi::addEventListener (handlerSession, "SESSION")) 
  {
    return(1);
  }
  
  xoc::hvmapi::ParamList aspect;
  aspect["EXECUTION_STATE"] = "DEVICE_CHANGE_COMPLETED";
  if (!xoc::hvmapi::addEventListener(handlerDeviceChangeComplete, "EXECUTION", aspect)) 
  {
    return(1);
  }

  xoc::hvmapi::ParamList aspect1;
  aspect1["EXECUTION_STATE"] = "TESTPROGRAM_DOWNLOAD_COMPLETED";
  if (!xoc::hvmapi::addEventListener(handlerTPLoadComplete, "EXECUTION", aspect1)) 
  {
    return(1);
  }

  xoc::hvmapi::ParamList aspect2;
  aspect2["EXECUTION_STATE"] = "TESTPROGRAM_EXEC_COMPLETED";
  if (!xoc::hvmapi::addEventListener(handlerTPComplete, "EXECUTION", aspect2)) 
  {
    return(1);
  }

  xoc::hvmapi::ParamList results; 
  cout << "FA: StartSmartestSession,any key to start one new session"
       << endl;    
  getchar();

  xoc::hvmapi::ParamList params, qrResults; 
  params["ONLINE_MODE"] = "OFFLINE";
  params["USER_MODE"] = "SUPERVISOR";
  errObj = xoc::hvmapi::operation("StartSmartestSession",results,params);
  if ( errObj.isSuccess() ) 
  {
    sSession = results["SESSION_ID"];
    cout << "the session ID currently started is " << sSession << endl;
  }

   errObj = xoc::hvmapi::query("PingSmartestSessions",qrResults,xoc::hvmapi::ParamList(),sSession);
      
   if (errObj.isSuccess())
  {
    cout<<"the session status is " << qrResults["LIFE_STATUS"] <<endl;
  } 
  else 
  {
    cout <<" error to ping session " << endl;
    cout <<" the code is " << errObj.code() <<endl;
    cout << "the text is " << errObj.text() << endl;
  }
  
  cout << "FA: eventLoop started, now go to the evenloop" << endl;
  xoc::hvmapi::eventLoop();
  cout << "FA: eventLoop finished" << endl;

  cout << "now the main thread is executed again, any key to quit the client" << endl;
  getchar();
  
  return (0);  
}

Example of monitoring the production status

#include "xoc/hvmapi/hvmapi.hpp"
#include "xoc/hvmapi/Status.hpp"
#include <iostream>
#include <cstdio>
#include <unistd.h>

using namespace std;
std::string sSession;

void globalEventHandler(const std::string &type, const xoc::hvmapi::ParamList &params,
                        unsigned long sec, long usec, const std::string &session)
{

  if ( type == "SESSION" )
  {
    string  sessionState = params["SESSION_STATE"];
    cout << "PM: Event received ### Session: " << session << " # EventType: " << type 
         << " # State: " << sessionState << endl;
  }
  else if ( type == "EXECUTION")
  {
    string execState = params["EXECUTION_STATE"];
    cout << "PM: Event received ### Session: " << session << " # EventType: " << type 
         << " # State: " << execState << endl;  
  
  }
  else if ( type ==  "ALARM" )
  {
    string severity = params["ALARM_SEVERITY"];
    string code = params["ALARM_CODE"];
    string text = params["ALARM_TEXT"];
    cout << "PM: Event received ### Session: " << session << " # EventType: " << type
         << " # Severity: " << severity << "# Code: " << code << " # Text: " << text << endl;
  }
  else
  {
    cout << "ERROR, receive other event category: " << type << endl;
  }
}        

int main(int argc, char **argv) 
{
  xoc::hvmapi::ParamList aspects;
  xoc::hvmapi::addEventListener(globalEventHandler, "SESSION");  
  xoc::hvmapi::addEventListener(globalEventHandler, "ALARM");  
  xoc::hvmapi::addEventListener(globalEventHandler, "EXECUTION");  
  xoc::hvmapi::eventLoop();
  return (0);  
}
