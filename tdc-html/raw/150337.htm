The EVENT_WAVEFORM_VC (Voltage Clamp) waveform class provides the functionality to define events that generate waveforms in the force current mode.

This class is supported only for the DC Scale XPS256 card in PMU operating mode.

Class definition

The class derives from the EVENT_BASE class.

The object that is created by this class has to be assigned to the PATTERN_CONTROLLER which is responsible for further handling.

This class is defined in the header file /opt/hp93000/soc/prod_com/include/MAPI/Events.h.

Constructor
EVENT_WAVEFORM_VC (const STRING & id);
EVENT_WAVEFORM_VC (INT id);

The constructor creates an object of the type EVENT_WAVEFORM_VC and assigns the given identifier to the object.

Parameters
Parameter Description
id

Identifier of the event.

The identifier may be defined as integer value or as string.
Return value
The constructor has no return value.

Example

The following example creates a new DC event, fills the DC event with the data of two ANALOG_WAVEFORM objects and inserts the DC event into an already existing pattern. The programmed data are then read back from the event object and shown in the console window.

PATTERN_CONTROLLER patController;

ON_FIRST_INVOCATION_BEGIN();
{
    //
    // Create two waves for the lower and upper clamp voltage
    //
    ANALOG_WAVEFORM sineWave_low    ("sine_low");
    ANALOG_WAVEFORM sineWave_high   ("sine_high");
    INT             sineWaveSamples = 20;

    sineWave_low
        .definition (TM::SINE)
        .amplitude  (2.0 V)
        .offset     (-3.0 V)
        .samples    (sineWaveSamples)
        .periods    (2);

    sineWave_high
        .definition (TM::SINE)
        .amplitude  (4.0 V)
        .offset     (5.0 V)
        .samples    (sineWaveSamples)
        .periods    (2);

    //
    // Create a new event object
    //
    EVENT_WAVEFORM_VC event_WAVEFORM_VC ("WAVEFORM_VC");

    //
    // Set the event data
    //
    event_WAVEFORM_VC
        .vClampLow              (sineWave_low)
        .vClampHigh             (sineWave_high)
        .sampleVectorDistance   (50);

    //
    // Insert the event into a pattern
    //
    PATTERN & pattern = patController.label("...", "@");

    pattern.vecEvents("pin1").clearAll();

    pattern.vecEvents("pin1")
        .insert(100, event_WAVEFORM_VC)
        .setup ();

    //
    // Read data from the event object
    //
    ARRAY_D readWave_low, readWave_high;
    INT     samples, distance, repeats;

    readWave_low  = event_WAVEFORM_VC.getVClampLow();
    readWave_high = event_WAVEFORM_VC.getVClampHigh();

    cout << "wave size low  = " << readWave_low.size() << endl;
    cout << "wave size high = " << readWave_high.size() << endl;
    cout << endl;

    for (INT arrayIndex = 0; arrayIndex < readWave_low.size(); arrayIndex++)
    {
        cout << "readWave_low[" << arrayIndex << "] = " << readWave_low[arrayIndex] << endl;
    }

    cout << endl;

    for (INT arrayIndex = 0; arrayIndex < readWave_high.size(); arrayIndex++)
    {
        cout << "readWave_high[" << arrayIndex << "] = " << readWave_high[arrayIndex] << endl;
    }

    event_WAVEFORM_VC.getSamples                (samples);
    event_WAVEFORM_VC.getSampleVectorDistance   (distance);
    event_WAVEFORM_VC.getRepeatCount            (repeats);

    cout << endl;
    cout << "samples  = " << samples  << endl;
    cout << "distance = " << distance << endl;
    cout << "repeats  = " << repeats  << endl;
}
ON_FIRST_INVOCATION_END();
