phHookActionStart

Subtopics

  1. Parameters:
  2. User-Specific Code:

This hook function is called (if activated, see Activating a Hook Function) on every PROB_HND_CALL from the application model, before the driver has performed any significant action. (The only actions of the driver before the call of the hook function are the check that the PROB_HND_CALL was valid in the position of the application model where it occurred, and the logging of the PROB_HND_CALL.)

phHookError_t phHookActionStart(
    phFrameAction_t call/* the identification of an
        incoming call from the
        application model */,
    phEventResult_t *result /* the result of this hook call */,
    char *parm_list_input /* the parameters passed over */,
        from the application model */,
    int parmcount/* the parameter count */,
    char *comment_out/* the comment to be send back */
        to the application model */,
    int *comlen/* the length of this comment */,
    phTcomUserReturn_t *stReturn  /* SmarTest return value, 
        if <result> is set to
        PHEVENT_RES_ABORT by this
        hook function */
)
{
    /* do checks and trace the call first */
    CheckInitialized(&hook);
    phLogHookMessage(hook.myLogger, PHLOG_TYPE_TRACE, 
        "phHookActionStart(0x%08lx,P%p,\"%s\",%d,\"%s\",%d,P%p)",
        (long) call, result, 
        parm_list_input ? parm_list_input : "(NULL)", parmcount, 
        comment_out ? comment_out : "(NULL)", comlen, 
        stReturn);

#ifdef ACTION_START_IMPLEMENTED

    /* Here is a sample implementation which prints out */
    /* a message on each ph_device_start and ph_device_done */
    /* call from the application model*/

    switch (call)
    {
      case PHFRAME_ACT_DEV_START:

    phLogHookMessage(hook.myLogger,
        PHLOG_TYPE_MESSAGE_0,
        "start of ph_device_start recognized by hook function");

    /* set the result value:
    the framework must continue with the ph_device_start
    actions */
    *result = PHEVENT_RES_CONTINUE;
    break;

      case PHFRAME_ACT_DEV_DONE:

    phLogHookMessage(hook.myLogger,
        PHLOG_TYPE_MESSAGE_0,
        "start of ph_device_done recognized by hook function");

    /* set the result value:
       the framework must continue with the ph_device_done
      actions */
    *result = PHEVENT_RES_CONTINUE;
    break;

      default:

    /* set the result value:
       the hook function has done nothing */
    *result = PHEVENT_RES_VOID;
    break;
    }

    return PHHOOK_ERR_OK;

#else

    /*  hook was not activated */
    return PHHOOK_ERR_NA;

#endif
}
Note For all type definitions, see the header files corresponding to the respective type. The header files are located in the directory ... /GenericHandler/include/ and in ... /GenericProber/include For example, for the type phTcomUserReturn_t see the header file phTcom.h
