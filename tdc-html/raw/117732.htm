DC event example: Threshold hysteresis test

Setup Function

void threshold_hyst_dyn_setup()
{
  cout << "threshold_hyst_dyn_setup_2()" << endl;

  ANALOG_WAVEFORM rampUpDown("rampUpAndDown");
  rampUpDown.definition(TM::RAMP)
            .min(2.5).max(3.5)
            .direction(TM::POS)
            .samples(11).periods(1);

  ANALOG_WAVEFORM rampDown("rampDown");
  rampDown.definition(TM::RAMP)
          .min(2.5).max(3.5).
           direction(TM::NEG)
          .samples(11).periods(1);

  rampUpDown.add(rampDown);

  PUT_DEBUG("LM393_IN_NEG", "LM393_THRESHOLD_HYST", rampUpDown.getArray());

  EVENT_WAVEFORM_VF eventRampUpDown("eventRampUpDown");
  eventRampUpDown.vForce(rampUpDown)
  eventRampUpDown.sampleVectorDistance(2000);
  PATTERN_CONTROLLER pc;

  pc.label("LM393_THRESHOLD_HYST")
    .vecEvents("LM393_IN_NEG")
    .insert(100, eventRampUpDown)
    .setup();
}

Execution Function

void threshold_hyst_dyn()
{
  ostringstream query;
  FirmwareTask  task;
  smatch        matches;

  static const STRING testName("Threshold Hysteresis (new dynamic Version)");

  INT logToReportWindow = 0;
  GET_TESTFLOW_FLAG("debug_analog", &logToReportWindow);

  ON_FIRST_INVOCATION_BEGIN();
    if ( logToReportWindow ) {
      cout << "Testing " << testName << "..." << endl;
    }

    CONNECT();

 
    // Store error map in RAM; not in the test processor.
    task.eval("DCRT RAM,(@@)\n");

    FUNCTIONAL_TEST();
    Boolean passed = GET_FUNCTIONAL_RESULT("LM393_OUT");
    if ( passed ) {
      throw runtime_error("Functional test has passed (must always fail).");
    }
  ON_FIRST_INVOCATION_END();

  task.eval("VALD? ERRM,(LM393_OUT)\n");
  if (! task.matchLine("VALD CONS,ERRM,([0-9]+),([0-9]+),\\(LM393_OUT\\)", &matches)) {
    string line;
    task.getLine(&line);

    ostringstream what;
    what << "Failed to read valid data range (bad firmware response: '"
         << line << "')";
    throw runtime_error(what.str());
  }

  int startCycle   = lexical_cast<int>(matches[1]);
  int numberCycles = lexical_cast<int>(matches[2]);

  query << "ERMP? ERRM," << startCycle << "," << numberCycles << ",,(LM393_OUT)" << endl;
  task.eval(query);

  string ermpData;
  task.getData(&ermpData);

  if ( numberCycles != ermpData.length() * 8 ) {
    throw runtime_error("Error map query returned wrong number of cycles.");
  }

  unsigned int firstTransition  = 0;
  unsigned int secondTransition = 0;
  findTransitionsInErrorMap(ermpData, firstTransition, secondTransition);

  PATTERN_CONTROLLER pc; 
  PATTERN& pattern = pc.label("LM393_THRESHOLD_HYST");
  DOUBLE riseThreshold = 0.0;
  if (! pattern.getForceVoltage("LM393_IN_NEG", firstTransition, riseThreshold)) {
    throw runtime_error("Could not determine rise threshold.");
  }

  DOUBLE fallThreshold = 0.0;
  if (! pattern.getForceVoltage("LM393_IN_NEG", secondTransition, fallThreshold)) {
    throw runtime_error("Could not determine fall threshold.");
  }
  
  DOUBLE hysteresis = riseThreshold - fallThreshold;
  
  if ( logToReportWindow ) {
    cout << "Site: " << CURRENT_SITE_NUMBER() << endl << endl;
  }

  judge_and_log_threshold_hyst(testName,
                               riseThreshold, "riseThresholdLimit_V",
                               fallThreshold, "fallThresholdLimit_V",
                               hysteresis, "hysteresisLimit_V",
                               logToReportWindow);
}

Result Post Processing (Transition Lookup)

void findTransitionsInErrorMap(const string& ermpData,
                               unsigned int& firstTransition,
                               unsigned int& secondTransition)
{
  enum SearchState { NoTransitionFound, FirstTransitionFound, SecondTransitionFound };

  SearchState searchState = NoTransitionFound;

  unsigned char c = ermpData[0];
  bool previousCycleFailed = c & 0x80;

  firstTransition  = 0;
  secondTransition = 0;

  unsigned int numberCycles = ermpData.length() * 8;

  for ( unsigned int currentCycle = 0;
        searchState != SecondTransitionFound && currentCycle < numberCycles;
        ++currentCycle )
  {
    unsigned char c = ermpData[currentCycle / 8];
    unsigned char d = 0x80 >> (currentCycle % 8);
    bool currentCycleFailed = c & d;

    switch ( searchState )
    {
    case NoTransitionFound :
      if ( currentCycleFailed != previousCycleFailed )
      {
        firstTransition = currentCycle;
        searchState = FirstTransitionFound;
        previousCycleFailed = ! previousCycleFailed;
      }
      break;
    case FirstTransitionFound :
      if ( currentCycleFailed != previousCycleFailed )
      {
        secondTransition = currentCycle;
        searchState = SecondTransitionFound;
      }
      break;
    default :
      assert(!"We should never get here.");
    }
  }
}
