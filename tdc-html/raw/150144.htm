The EVENT_CFVM (Current Force / Voltage Measurement) is a combined force current / voltage measure event class. This class provides the combined functionality of setting and measurement events in a single event.

This class is supported only for the DC Scale XPS256 card in PMU operating mode.

Class definition

The class derives from the EVENT_BASE class.

The object that is created by this class has to be assigned to the PATTERN_CONTROLLER which is responsible for further handling.

This class is defined in the header file /opt/hp93000/soc/prod_com/include/MAPI/Events.h.

Constructor
EVENT_CFVM (const STRING & id);
EVENT_CFVM (INT id);

The constructor creates an object of the type EVENT_CFVM and assigns the given identifier to the object.

Parameter Description
id

Identifier of the event.

The identifier may be defined as integer value or as string.
Return value
The constructor has no return value.

Example

The following example creates three new DC events, sets the data of the DC events and inserts the DC events into an already existing pattern. The programmed data are then read back from the event objects and shown in the console window.

event_CFVM_1 uses member functions limitLow() and limitHigh() to define the limits.

event_CFVM_2 uses the class LIMIT and the member function limits() to define the limits.

event_CFVM_3 uses member functions limitLow() and limitHigh() to define the limits but the limits are cleared by member function clearLimits() to demonstrate how this function works.

PATTERN_CONTROLLER patController;

ON_FIRST_INVOCATION_BEGIN();
{
    //
    // Create new event objects
    //
    EVENT_CFVM event_CFVM_1 ("CFVM_1");
    EVENT_CFVM event_CFVM_2 ("CFVM_2");
    EVENT_CFVM event_CFVM_3 ("CFVM_3");

    //
    // Set the event data
    //
    event_CFVM_1
        .cForce             (30.0 mA)
        .averages           (50)
        .outOfRangeCheck    (true)
        .clampChecks        (true, false)
        .limitLow           (15.0 mA)
        .limitHigh          (20.0 mA);

    LIMIT event_CFVM_2_limits;

    event_CFVM_2_limits.low     (TM::GE, 17.0 mA);
    event_CFVM_2_limits.high    (TM::LE, 22.0 mA);
    event_CFVM_2_limits.unit    ("mA;-3;mA;-3");

    event_CFVM_2
        .cForce             (32.0 mA)
        .averages           (50)
        .outOfRangeCheck    (true)
        .iClampChecks       (false, true)
        .limits             (event_CFVM_2_limits);

    event_CFVM_3
        .limitLow           (15.0 mA)   // Cleared with clearLimits()
        .limitHigh          (20.0 mA)   // Cleared with clearLimits()
        .clearLimits        ();

    //
    // Insert the events into a pattern
    //
    PATTERN & pattern = patController.label("...", "@");

    pattern.vecEvents("pin1").clearAll();

    pattern.vecEvents("pin1")
        .insert(100, event_CFVM_1)
        .insert(150, event_CFVM_2)
        .insert(200, event_CFVM_3)
        .setup ();

    //
    // Read data from the event object
    //
    EVENT_VALUE_DOUBLE  forceValue_1, lowLimit_1, highLimit_1;
    EVENT_VALUE_INT     averages_1;
    bool                lowLimitDefined_1, highLimitDefined_1, outOfRangeCheck_1;
    bool                clampCheckLow_1, clampCheckHigh_1;

    event_CFVM_1.getCForce          (forceValue_1);
    event_CFVM_1.getAverages        (averages_1);
    event_CFVM_1.getOutOfRangeCheck (outOfRangeCheck_1);
    event_CFVM_1.getClampChecks     (clampCheckLow_1, clampCheckHigh_1);

    lowLimitDefined_1  = event_CFVM_1.getLimitLow   (lowLimit_1);
    highLimitDefined_1 = event_CFVM_1.getLimitHigh  (highLimit_1);

    cout << "forceValue_1          = " << forceValue_1                                      << endl;
    cout << "averages_1            = " << averages_1                                        << endl;
    cout << "outOfRangeCheck_1     = " << outOfRangeCheck_1                                 << endl;
    cout << "clampCheckLow_1       = " << clampCheckLow_1                                   << endl;
    cout << "clampCheckHigh_1      = " << clampCheckHigh_1                                  << endl;
    cout << "lowLimit_1  (defined) = " << lowLimit_1  << " (" << lowLimitDefined_1  << ")"  << endl;
    cout << "highLimit_1 (defined) = " << highLimit_1 << " (" << highLimitDefined_1 << ")"  << endl;
    cout << endl;

    EVENT_VALUE_DOUBLE  lowLimit_2, highLimit_2;
    bool                lowLimitDefined_2, highLimitDefined_2;

    lowLimitDefined_2  = event_CFVM_2.getLimitLow   (lowLimit_2);
    highLimitDefined_2 = event_CFVM_2.getLimitHigh  (highLimit_2);

    cout << "lowLimit_2  (defined) = " << lowLimit_2  << " (" << lowLimitDefined_2  << ")"  << endl;
    cout << "highLimit_2 (defined) = " << highLimit_2 << " (" << highLimitDefined_2 << ")"  << endl;
    cout << endl;

    EVENT_VALUE_DOUBLE  lowLimit_3, highLimit_3;
    bool                lowLimitDefined_3, highLimitDefined_3;

    lowLimitDefined_3  = event_CFVM_3.getLimitLow   (lowLimit_3);
    highLimitDefined_3 = event_CFVM_3.getLimitHigh  (highLimit_3);

    cout << "lowLimit_3  (defined) = " << lowLimit_3  << " (" << lowLimitDefined_3  << ")"  << endl;
    cout << "highLimit_3 (defined) = " << highLimit_3 << " (" << highLimitDefined_3 << ")"  << endl;
}
ON_FIRST_INVOCATION_END();
