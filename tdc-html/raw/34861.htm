Example of yield monitor

The following C program is an example of a simple, stand-alone data retrieval process built on the functions and data types of the data retrieval library. The program retrieves all events of the BinReachedEventType type and calculates the current yield after each device is binned.

Using the information in Data Retrieval Library, the implementation recommendations given in the previous section, and this example, you will be able to create your own data retrieval programs or functions and make full use of the capabilities of event data logging.

/*
 * ******************************************************************* 
 *
 * File: yieldmon.c 
 * Description: This is a yield monitor serving as usage example for the 
 *              DRL (Data Retrieval Library).
 *
 * Compile with:
 *
 * cc -Ae -g -I/opt/hp93000/soc/com/include -L/opt/hp93000/soc/com/sh_lib
 * -oyieldmon yieldmon.c -ldrl -lmc_macc
 *
 *
 * Author:  25-Nov-97 Gerd Bleher
 * Changes: 21-Sep-98 Andreas Schiffer 
 *
 * ******************************************************************/ 
     
#include <stdio.h>
#include <unistd.h>
     
#include "drl.h"  /* the header file of the data retrieval library */
     
     
/*
 * The global variables <good_parts> and <tested_parts> are used as 
 * tallies for the yield calculation.
 */
     
int good_parts, tested_parts;
     
/*
 * The global variable <my_stream> holds the DRLDataStream from which 
 * the events are read.
 */

DRLDataStream my_stream;          
     
/*
 * ******************************************************************* 
 *
 * SIGNATURE
 *
 *   void ProcessBinReachedEventAttributes(const DRLEventRec *event) 
 *
 * DESCRIPTION
 *
 *   Processes the attributes of the BinReached event <event>, 
 *   that is, it tallies the tested devices, calculates the yield 
 *   and prints the yield statistics. 
 *
 * RETURN VALUE
 *
 *   None
 *
 * AUTHOR
 *   25-Nov-97 Gerd Bleher
 * ******************************************************************* 
 */

void ProcessBinReachedEventAttributes(const DRLEventRec *event) 
{
  int
    number_of_attributes,
    i;
  double
    yield;
  DRLSiteIdAttr
    site_id;
  DRLAttributeRec
    attribute_rec;
  DRLBinAttr
    bin_attribute;
     
  number_of_attributes = DRLEventGetNoOfAttributes(event); 
  for (i=0; i<number_of_attributes; i++)
    {
      if (DRLEventGetAttribute(event,i,&attribute_rec))
        {
          fprintf(stderr,"Error while reading attributes ");
          fprintf(stderr," - skipping event.\n"); 
          return;
        }
     
      switch (attribute_rec.type)
        {
        case BinAttrType:
          bin_attribute = attribute_rec.data.bin_attr; 
          break;
     
        case SiteIdAttrType:
          site_id = attribute_rec.data.site_id_attr; 
          break;
        }
    }
     
  tested_parts++;
  if (bin_attribute->is_good_bin)
    {
      good_parts++;
    }
     
  yield = ((double)good_parts) / (double)tested_parts; 
     
  fprintf(stderr,"Site %d binned to bin %2s(%2d). Yield = %d/%d = %.2f%%\n",
          (int)site_id,
          bin_attribute->bin_name,
          bin_attribute->bin_number,
          good_parts,
          tested_parts,
          100*yield);
}
     
     
/*
 * ******************************************************************* 
 *
 * SIGNATURE
 *
 *   void Init(void)
 *
 * DESCRIPTION
 *
 *   Attaches to the system data log stream and initializes the active 
 *   event mask to "no event masked".
 *
 * RETURN VALUE
 *
 *   None
 *
 * AUTHOR
 *   25-Nov-97 Gerd Bleher
 * ******************************************************************* 
 */

void Init(void)
{
  good_parts = 0;
  tested_parts = 0;
     
  DRLStart(0, (DRL_SYNC_COUNTER | DRL_RESET_MASK | DRL_WAIT_FOR_EVENT), &my_stream);
  /*
   * Attaches to the system data log stream. Only events logged from now on 
   * are received, the active event mask is set to "no event masked".
   */
}
     
     
/*
 * ******************************************************************* 
 *
 * SIGNATURE
 *
 *   void SetActiveEventMask(void)
 *
 * DESCRIPTION
 *
 *   Sets a new active event mask so that all events are masked 
 *   except for the BinReached event.
 *
 * RETURN VALUE
 *
 *   None
 *
 * AUTHOR
 *   25-Nov-97 Gerd Bleher
 * ******************************************************************* 
 */

void SetActiveEventMask(void)
{
  DRLEventType
    event_type;
  DRLEventMask
    my_event_mask;
     
  for (event_type=0; event_type<NumberOfEventTypes; event_type++) 
    {
      my_event_mask.event_mask[event_type] = 1;
    }
     
  my_event_mask.size = NumberOfEventTypes; 
  /*
   * All events are masked and the event mask size is set correctly. 
   */
     
  my_event_mask.event_mask[BinReachedEventType] = 0; 
  DRLSetEventMask(&my_event_mask, &my_stream);
  /*
   * The BinReached event is unmasked and the active event mask is set as 
   * specified in my_event_mask
   */
}
     
     
/*
 * ******************************************************************* 
 *
 * SIGNATURE
 *
 *   void EventLoop(void)
 *
 * DESCRIPTION
 *
 *   Waits for an event and branches to the appropriate event handler. 
 *
 * RETURN VALUE
 *
 *   None
 *
 * AUTHOR
 *   25-Nov-97 Gerd Bleher
 * ******************************************************************* 
 */

void EventLoop(void)
{
  DRLEventRec
    event_rec;
     
  printf("Press \'Ctrl-c\' to stop the yield monitor.\n");
     
  while (!DRLGetNextUnmaskedEvent(&my_stream, &event_rec))
  /*  Event Loop  */
  {            /* Event Processing  */
    switch (DRLEventGetType(&event_rec)) 
    {
      case BinReachedEventType:
        /* This is the only event we're interested in */ 
        ProcessBinReachedEventAttributes(&event_rec);  
                                /* Attribute Processing */
        break;
     }
     /* Errors of DRLGetNextUnmaskedEvent are ignored */
   } /* End of Event Loop */
}
     
main()
{
  Init();
  SetActiveEventMask();
  EventLoop();
  DRLStop(&my_stream);     /* End Data Retrieval */
}

Appendix B: The DRLDump Tool describes a simple data retrieval program capable of reading and processing all existing types of events and attributes. This tool is a very good example of data retrieval. We recommended that you use the source code of this tool as a template for writing your own data retrieval function.

The Event Formatter described in Using the Event Formatter uses the same mechanisms for retrieving data log data as the yield monitor above. Except for the differences outlined in Important notes for user-written report formatters, the formatter is implemented as recommended in this section.

Because of its complexity, the Event Formatter is not particularly suitable as a sample data retrieval implementation. However, the formatter demonstrates well that the DRL allows for programming of sophisticated data retrieval processes.
