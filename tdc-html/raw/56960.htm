Example of an equation based level setup

The following example shows an equation based level setup.

EQNSET 1 "Equation set 1"

SPECS

vcc_level       [V]  # Base Vcc voltage
vcc_offset      [V]  # Offset to add to vcc_level
aclk_vil        [V]  # Initial VIL for aclk pins
aclk_vi_swing   [V]  # VIL-VIH swing for aclk pins
aclk_vo_swing   [V]  # VOL-VOH swing for aclk pins
adata_vil       [V]  # Initial VIL for adata pins
adata_vi_swing  [V]  # VIL-VIH swing for adata pins
adata_vo_swing  [V]  # VOL-VOH swing for adata pins

EQUATIONS

aclk_vih  = aclk_vil + aclk_vi_swing    # VIH for aclk
                                        # pins
aclk_vol  = aclk_vil + 1.5              # VOL for aclk
                                        # pins
aclk_voh  = aclk_vol + aclk_vo_swing    # VOH for aclk
                                        # pins
adata_vih = adata_vil + adata_vi_swing  # VIH for adata
                                        # pins
adata_vol = adata_vil + 1.5             # VOL for adata
                                        # pins
adata_voh = adata_vol + adata_vo_swing  # VOH for adata
                                        # pins

DPSPINS vcc
vout   = vcc_level + vcc_offset
ilimit = 20
t_ms   = 0
offcurr= min

LEVELSET 1

PINS aclkout
vil= aclk_vil
vih= aclk_vih
vol= aclk_vol
voh= aclk_voh

PINS adata
vil= adata_vil
vih= adata_vih
vol= adata_vol
voh= adata_voh

Functional changes

  1. Introduced before SmarTest 6.3.2
