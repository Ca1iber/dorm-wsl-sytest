Lets you create a source-synchronous test method and execute this test without using the MTP provided source-synchronous test method.

For the description of the source-synchronous test, see source-synchronous test method.

This API is defined in the header file /opt/93000mts/mts/solution_tools/common/include/SOURCESYNC_TEST.h.

Syntax

SOURCESYNC_TEST();

This constructor creates an object for the SOURCESYNC_TEST class.

List of parameters

There are no parameters.

Return value

There is no return value.

Example

Use the SOURCESYCE_TEST API to create a sourcsync test.

#include "solution_tools.h"
#include "testmethod.hpp"
#include "mapi.hpp"
class CSourceSyncTest: public testmethod::TestMethod
{
  virtual void run()
  {
    //Create a source sync test object.
    SOURCESYNC_TEST ssTest;
    // The following block only executes for the first site of multi-site setup
    ON_FIRST_INVOCATION_BEGIN();
    {
      // hand over the parameters to the test class.
      ssTest.startCycle("1;200")
            .stopCycle("10;220")
            .testSet("Min Value,Max Value,Mean Value,Sigma Value");
      // Create new test with name "Setup" and initialize it.
      ssTest.test("Setup")
            .edgeList("1,3,5,7")
            .specVar("tSetup")
            .sweepSpecVar("tsweep_a")
             .sweepStart("0.5")
             .sweepStop("0.7")
             .sweepStepSize(0.1);
      // Create new test with name "Hold" and initialize it.
      ssTest.test("Hold")
            .edgeList("2,4,6,8")
            .specVar("tHold")
            .sweepSpecVar("tsweep_b")
            .sweepStart("2")
            .sweepStop("2.5")
            .sweepStepSize(0.1);
      // Create new domain with name "domainA" and initialize it.
      ssTest.domain("domainA")
            .pins("DQA0_PS,DQA1_PS,DQA2_PS,DQA3_PS,DQS_A_p_PS,DQS_A_n_PS")
            .bufferPins("DQA0_DUT,DQA1_DUT,DQA2_DUT");
      // Create new domain with name "domainB" and initialize it.
      ssTest.domain("domainB")
            .pins("DQB0_PS,DQB1_PS,DQB2_PS,DQB3_PS,DQS_B_p_PS,DQS_B_n_PS")
            .bufferPins("DQB0_DUT,DQB1_DUT,DQB2_DUT");
      // execute the test
      ssTest.execute();
    }
    ON_FIRST_INVOCATION_END();

  // set result
  ssTest.judgeAndLog();
  // Print out in the UI Report Window.
  ssTest.print();
  // Print out specified testNameA in UI Report Window.
  ssTest.test("Hold").print();
  // Get statistic result of sepecifie test and domain.
  double min;
  min = ssTest.test("Setup").domain("domainA").getMin();
  // User can extend HANDLER and use it to handle user-specified case.
  SS_PRINT_HANDLER handler;
  ssTest.handle(handler);
  // User can extend HANDLER and use it to handle user-specified case of specified test.
  }
};
