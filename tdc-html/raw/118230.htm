How to hide the operation control panel

If you Do not want the operators to touch the device setting, you can hide the operation control panel using the Linux system functions.

About this task

The window name of the operation control panel is "operation control".

Before you begin

Be familiar with the system functions to deal with the X-window, including XOpenDisplay(), DefaultScreen(), XDefaultRootWindow() and XWithdrawWindow().

To successfully build the functions, you need to add #include <X11/Xatom.h> and #include <X11/Xlib.h> in the file. You also need to add the option -I/usr/X11R6/include when compiling the source file and add the option -L/usr/X11R6/lib -lX11 when linking the object file.

Procedure

  1. Retrieve the current display screen.
  2. Hide the operation control panel with the XWithdrawWindow() function.

Results

The operation control panel was hided.

Example

The following example hides the operation control panels. findname() is a function defined to find the window to be hidden.

#include <X11/Xatom.h>
#include <X11/Xlib.h>
void hide_op_window()
{
  std::string error_message;
  std::string response;

  Display *dpy;
  int screen;
  Window *found_win;
  char search_win_name[18];
  int status, nfound, i;

  if((dpy=XOpenDisplay(0))==0)
  {
    cout <<"cannot open X display server:" <<endl;
    return;
  }
  screen=DefaultScreen(dpy);

  strcpy(search_win_name,"Operation Control");
  found_win = (Window *) calloc(1, sizeof(Window));
  if(findname(dpy,XDefaultRootWindow(dpy),search_win_name,found_win,1,&nfound)!=False)
  {
    for(i=0;i<nfound;i++)
    {
      status = XWithdrawWindow(dpy,found_win[i],screen);
    }
  }
  XCloseDisplay(dpy);
  return;
}

#define MAX_WIN_AT_LEVEL 25000
int findname( Display *dpy,
    Window top_win,
    char search_win_name[],
    Window win_found[],
    int win_found_max,
    int *nfound)
{
  Window buf1[MAX_WIN_AT_LEVEL];
  Window buf2[MAX_WIN_AT_LEVEL];
  Window *parents;
  Window *children;
  Window *tmp;
  int nparents;
  int nchildren;
  int parent;

  Window q_root_ret;
  Window q_parent_ret;
  Window *q_children_ret;
  unsigned int q_nchildren_ret;
  unsigned int q_child;
  int status;
  Atom actual_type;
  int actual_format;
  unsigned long nitems,leftover;
  unsigned char *data;

  parents = buf1;
  children = buf2;
  parents[0] = top_win;
  nparents = 1;
  *nfound = 0;
  while(nparents>0)
  {
    /*check name*/
    for(parent=0;parent<nparents;parent++)
    {
      status=XGetWindowProperty(dpy,parents[parent],XA_WM_NAME,0L,80L,False,
                                XA_STRING,&actual_type,&actual_format,
                                &nitems, &leftover, &data);
      if (status == Success)
      {
        /* printf("name: <%s>\n", (char *) data); */
        if ((char *) data != NULL)
          {
            if (strcmp((char *) data,search_win_name) == 0 )
            {
              XFree(data);
              win_found[*nfound]=parents[parent];
              (*nfound)++;
              if (*nfound >= win_found_max)
              {
                return True;
              }
            }
            XFree(data);
          }
        }
        else
        {
          printf("XGetWindowProperty failed. aborting.\n");
          return -1;
        }
    }
    /*build children list*/
    nchildren = 0;
    for(parent=0;parent<nparents;parent++)
    {
      XQueryTree(dpy,parents[parent],&q_root_ret,&q_parent_ret,
                 &q_children_ret,&q_nchildren_ret);
      if ( (nchildren + q_nchildren_ret) > MAX_WIN_AT_LEVEL)
      {
        printf("MAX_WIN_AT_LEVEL (%d) exceeded. aborting.\n",MAX_WIN_AT_LEVEL);
        return -1;
      }
      for(q_child=0;q_child<q_nchildren_ret;q_child++)
      {
        children[nchildren] = q_children_ret[q_child];
        nchildren++;
      }
      XFree(q_children_ret);
    }

    /*swap children to parents*/
    tmp = parents;
    parents = children;
    nparents = nchildren;
    children = tmp;
    nchildren = 0;
  }
  if (*nfound > 0)
    return True;
  else
    return False;
}
