How to use session info API

This section describes how to use the session info API.

About this task

The client using HVM session info API is independent of SmarTest versions.

Before you begin

Get the HVM session info API release package hvmapi-1.0.0-0.el7.x86_64.rpm.

Procedure

  1. To install the HVM session info API, change to the directory where the release package locates and enter the following command in the command line:

    $ rpm -i hvmapi-1.0.0-0.el3.i386.rpm

  2. Create your daemon C/C++ program with the APIs.

    You can find the example program in the directory /opt/hp93000/soc_hvmapi/examples.

  3. Build your client program like any ordinary C++ program. Use a makefile or other tools to build the program. For example, you use a makefile, change to your program directory and type in make in the Linux command line to build your program.

    You can find the example makefile in the directory /opt/hp93000/soc_hvmapi/examples.

  4. Execute the client.

Results

The client will monitor the SmarTest session information and execute the functions defined.

Example

The following code reports the current active SmarTest session.

#include <xoc/hvmapi/hvmapi.hpp>"
#include <iostream>
#include <unistd.h>
 
using namespace std;
using namespace ::xoc::hvmapi;
 
 
int main(int argc, char **argv) 
{
 
  cout << "New Session Detector is running..." << endl << endl;
 
  ParamList outParams;
  Status retObj;
 
  vector<string> sessionIdList;
  vector<string> activeSessionIdList;
  vector<string> lastActiveSessionIdList;
  vector<string> lifeStatusList;
  bool isNewSession = true;
 
  // daemon runs 24x7
  while(true)
  {
    // list SmarTest session ids
    retObj = query("PingSmartestSessions", outParams);
    if(retObj.isSuccess())
    {
      sessionIdList = outParams.valueArray("SESSION_ID_LIST");
      lifeStatusList = outParams.valueArray("LIFE_STATUS_LIST");
 
      // filter the active session ID list
      activeSessionIdList.clear();
      for(unsigned int i=0; i<lifeStatusList.size(); i++)
      {
        if(lifeStatusList.at(i) == "ACTIVE")
        {
          activeSessionIdList.push_back(sessionIdList.at(i));
        }
      }
 
      // active session list change detect
      if(activeSessionIdList != lastActiveSessionIdList)
      {
        // display current session ID list each time session list changes
        if(!activeSessionIdList.empty())
        {
          cout << endl << "********************************************" << endl;
          cout << "current active smartset session id list:" << endl;
          for(vector <string>::iterator it=activeSessionIdList.begin();
              it!= activeSessionIdList.end(); it++)
          {
            cout << *it << endl;
          }
          cout << endl << "********************************************" << endl;
        }
        else
        {
          cout << endl << "********************************************" << endl;
          cout << "no active SmarTest session." << endl;
          cout << endl << "********************************************" << endl;
        }
 
        for(unsigned int i = 0; i < activeSessionIdList.size(); i++)
        {
          isNewSession = true;
          for(unsigned int j = 0; j < lastActiveSessionIdList.size(); j++)
          {
            if(activeSessionIdList[i] == lastActiveSessionIdList[j]) 
            {
              isNewSession = false;
              break;
            }
          }
          // display session profile for each new active session
          if(isNewSession)
          {
            cout << endl << "New session \"" <<activeSessionIdList.at(i)<<"\" detected." << endl;
            retObj = query("GetSmartestSessionProfile", outParams,
                           ParamList(), activeSessionIdList.at(i));
            if(retObj.isSuccess())
            {
              cout << "session \"" << activeSessionIdList.at(i)<<"\" profile:" << endl;
              cout << "version: " << outParams["VERSION"] << endl;
              cout << "patch_level: " << outParams["PATCH_LEVEL"] << endl;
              cout << "online_mode: " << outParams["ONLINE_MODE"] << endl;
              cout << "user_mode: " << outParams["USER_MODE"] << endl;
              cout << "login_user: " << outParams["LOGIN_USER"] << endl;
            }
            else
            {
              cout << "GetSmartestSessionProfile query faied" << endl;
              cout << "-info: "<<retObj.text() << endl;
            }
          }
        }// for each activeSessionIdList
      }
 
    lastActiveSessionIdList = activeSessionIdList;
    }
    else
    {
      cout << "PingSmartestSessions query faied" << endl;
      cout << "-info: "<<retObj.text() << endl;
    }
 
      usleep(200000);
 
  }// while(true)
 
  return 0;
 
}
