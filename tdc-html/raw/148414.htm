releaseInstance() - DigitalCapturePort API

This API is used to release the instance of the DigitalCapturePort.

This function is defined in the header file /opt/hp93000/soc/testmethod/include/CCTMultiportSync.hpp.

Syntax

void releaseInstance();

Example

CCTMultiportSync    Sync;
GET_TESTSUITE_NAME(tsName);
CCTtsName = CCTgetTestsuiteName();
Sync.registerCCTestMethod(tsName, this);

DigitalCapturePort::getInstance().setPort("DigCap_Port");
DigitalCapturePort::getInstance().setIdentifier(tsName);
Sync.registerSyncPort(DigitalCapturePort::getInstance());
    
Sync.start();
vectorVariable adcCap_lin(capture_variable);
DigitalCapturePort::getInstance().setDefaultPattern("D", 8);
    
adcCap_lin.pin("ADC_D7,ADC_D6,ADC_D5,ADC_D4,ADC_D3,ADC_D2,ADC_D1,ADC_D0")
        mode("PAR")
        initSkipSamples(0)
                .frameLengthInSamples(capSampleNum)
                .frameCount(1)
                .skipSamples(0)
                .interFrameSkipSamples(0);
DigitalCapturePort::getInstance().capture(adcCap_lin, "ADC_D7,ADC_D6,ADC_D5,ADC_D4,ADC_D3,ADC_D2,ADC_D1,ADC_D0",capSampleNum);

Sync.sync();
Sync.flush();
......
