Topic 151603

setStopSiteLoop()

The run() function of test suites is usually called for each active site. The setStopSiteLoop() API enables you to call the run() function on the first active site only to minimize the test time.

To use this API you need to meet these requirements:

  • Common thread in SmartCalc "Thread per site" and "Common thread"
  • Using SmartCalc (SMC) with SmartRDI
  • Hidden data logging with SmartCalc (SmartCalc framework)
  • Keep everything in the ON_FIRST_INVOCATION block

Syntax

void setStopSiteLoop (bool stop);
Parameter

Parameter

Description

stop

If set to true the test suite will not be called for the remaining sites.

Default: false

If this is set to true, but you want to go back to the default behavior, you need to set it to false again explicitly.
Return value
There is no return value.

Example

This example executes the ON_FIRST_INVOCATION block on the first active site only.

#include "testmethod.hpp"

#include "mapi.hpp"
using namespace std;

// Set the limits for online execution
static const LIMIT limitDpsPin(TM::GE,0 mA, TM::LE, 1 A);

class dcOnce: public testmethod::TestMethod {
private:
  STRING  testSuiteName;
  ARRAY_I activeSites;
  INT     offline;
  ARRAY_D measValues;
public:
  virtual void SMC_backgroundProcessing()
  {
    DC_RESULT_ACCESSOR result(testSuiteName);

    // Hidden data logging
    // The data logging depends if the test is executed online or offline
    for (INT siteIdx=0, siteEnd=activeSites.size(); siteIdx < siteEnd; siteIdx++) {
      if (offline)
         SMC_TEST(activeSites[siteIdx],"dps",testSuiteName,limitDpsPin, 1.0 A /activeSites[siteIdx],true,1);
      else {
         result.uploadResult(activeSites[siteIdx],"dps",TM::RESULT_INDEX, TM::RESULT_RANGE(TM::RESULT_INDEX,0,1));
         measValues = result.getValues("dps");
         SMC_TEST(activeSites[siteIdx],"dps",testSuiteName,limitDpsPin, measValues[0],true,1);
      }
    }
  }
protected:
  virtual void run()
  {
    ON_FIRST_INVOCATION_BEGIN();
      // initialization
      CONNECT();
      GET_TESTSUITE_NAME(testSuiteName);
      GET_SYSTEM_FLAG("offline",&offline);
      GET_ACTIVE_SITES(activeSites);

      // Enable the run once mechanism
      setStopSiteLoop(true);

      // define & select result area
      RESULT_AREA_CONFIG_SET resultArea(testSuiteName);
      resultArea.pin("dps").setDcValueResultAreaSize(1024);
      resultArea.setAsPrimary();
      
      // Execute the test
      SMC_FUNCTIONAL_TEST();
      // deselect result area for hidden upload
      resultArea.deselectGlobally();

      // start single thread background processing
      SMC_ARM();
    ON_FIRST_INVOCATION_END();

    return;
  }

  virtual void initialize()
  {
  }

  virtual void postParameterChange(const string& parameterIdentifier)
  {
    return;
  }

  virtual const string getComment() const 
  {
    string comment = "dcOnce";
    return comment;
  }
};
REGISTER_TESTMETHOD("dcOnce", dcOnce);
