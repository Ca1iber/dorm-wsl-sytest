  • Waveforms are linked to data uploads by SmartCalc result handles
  • No changes required in test methods using SmartCalc
  • Solution covers SmartCalc and Burst upload

Example

Setup file

<offlineResultEmulation>

   <measurementname="TxDAC_106"> 	
     <sitenumber="1">
       <pinname="TX0">
         <waveformid="1" file="/home/demo/MATLAB/tx106m1s1.txt"/>
         <waveform id="2" file="/home/demo/MATLAB/tx106m2s1.txt"/>
       </pin>
     </site>
     <sitenumber="2">
       <pinname="TX0">
         <waveform id="1" file="/home/demo/MATLAB/tx106m1s2.txt"/>
         <waveform id="2" file="/home/demo/MATLAB/tx106m2s2.txt"/>
       </pin>
     </site>
   </measurement>

</offlineResultEmulation>
