Topic 111198

How to use the test suite control API

About this task

To control the test suites with the test suite control APIs, you need to create a C++ program and build the program to create a client.

Before you begin

Make sure you are using SmarTest 6.5.2 or later versions.

You must build your program from within the run shell. To activate the run shell, execute the following command:

$/opt/hp93000/soc/system/bin/run -S

Procedure

  1. Create a C++ program to control the test suites with the test suite control APIs.

    Include the header file testsuitecontrol_api.hpp and use the name space testsuitecontrol in the program.

  2. Build the C++ program to create a client.
    For a 32-bit system, use the following compilation and link options:
    Note For a 64-bit system, use -m64 to replace -m32 or do not use the option at all.

    Compilation options: -I/opt/hp93000/soc/prod_com/include/ -m32

    The compiling command is like:

    g++ -I/opt/hp93000/soc/prod_com/include/ -m32 -c -o main.o main.cpp

    Link options: -m32 -L /opt/hp93000/soc/prod_com/sh_lib/ -Wl,-rpath /opt/hp93000/soc/prod_com/sh_lib/ -ltestsuitecontrol_api

    The linking command is like:

    g++ -m32 -L /opt/hp93000/soc/prod_com/sh_lib/ -Wl,-rpath /opt/hp93000/soc/prod_com/sh_lib/ -ltestsuitecontrol_api -o t1.exe. main.o

    For Red Hat 3, use the following compilation and link options:

    Compilation options: -I /opt/hp93000/soc/prod_com/include/ -I /opt/openoffice.org/OpenOffice.org2.0_SDK/include/stl/ -DGXX_INCLUDE_PATH=/usr/include/c++/3.2.3/

    The compiling command is like:

    g++ -I /opt/hp93000/soc/prod_com/include/ -I/opt/openoffice.org/OpenOffice.org2.0_SDK/include/stl/ -DGXX_INCLUDE_PATH=/usr/include/c++/3.2.3/ -c -o main.o main.cpp

    Link options: -L /opt/hp93000/soc/prod_com/sh_lib/ -Wl,-rpath /opt/hp93000/soc/prod_com/sh_lib/ -ltestsuitecontrol_api

    The linking command is like:

    g++ -L /opt/hp93000/soc/prod_com/sh_lib/ -Wl,-rpath /opt/hp93000/soc/prod_com/sh_lib/ -ltestsuitecontrol_api -o t1.exe main.o

  3. Execute the client to as below:

    /opt/hp93000/soc/system/bin/run -x user_executable_program

Examples

Executing an existing test suite

The following example executes an existing test suite in the current test flow.

#include "testsuitecontrol_api.hpp"
#include <iostream>
#include <stdexcept>
 
using namespace std;
using namespace testsuitecontrol;
 
int main(int argc, char** argv)
{
int nRet = -1;
 
try
{
initializeSession(); 
TestsuiteProxy ts = getTestsuite("mysuite");
//execute it on site 1,3,5
set<int> sites;
sites.insert(1);
sites.insert(3);
sites.insert(5);
setActiveSites(sites);
 
ts.executeActiveSites();
 
for(set<int>::const_iterator it = sites.begin(); it != sites.end(); ++it)
{ 
int siteNum = *it;
if (ts.isPass(siteNum))
{
cout<<"\"mysuite\" was executed on site"<<siteNum<<": PASSED."<<endl;
}
else
{
cout<<"\"mysuite\" was executed on site"<<siteNum<<": FAILED."<<endl;
}
}
detachSession();
nRet = 0;
}
catch(runtime_error& error)
{
cerr<<error.what()<<endl;
}
return nRet;
}

Create, Setup and execute a test suite

The following example creates a floating test suite, setup and execute the floating test suite.

#include "testsuitecontrol_api.hpp"
#include <iostream>
#include <stdexcept> 
 
using namespace std;
using namespace testsuitecontrol;
 
int main(int argc, char** argv)
{
int nRet = -1; 
try
{ 
//default session: the SmarTest session of the current display
//default UTM executor: the first one (i.e. TestMethodManager).
initializeSession(); 

TestsuiteProxy ts = createFloatingTestsuite("floating");
ts.attachTestmethod("ac_tml.AC_TEST.FunctionalTest");
 
//setup this testsuite
ts.setEquationBasedLevel("2","1","3");
ts.setEquationBasedTiming("3","2","2");
ts.setSequencerLabel("\"MATCH_LOOP\"");
 
//execute it on currently active site
ts.executeActiveSites();
set<int> sites = getActiveSites();
for(set<int>::const_iterator it = sites.begin(); it != sites.end(); ++it)
{ 
int siteNum = *it;
if (ts.isPass (siteNum))
{
cout<<"\"mysuite\" was executed on site"<<siteNum<<": PASS."<<endl;
}
else
{
cout<<"\"mysuite\" was executed on site"<<siteNum<<": FAILED."<<endl;
}
}
detachSession();
nRet = 0;
} 
catch(runtime_error& error)
{
cerr<<error.what()<<endl;
}
 
return nRet;
}

Delete a test suite

The following example deletes a floating test suite.

#include "testsuitecontrol_api.hpp"
#include <iostream>
#include <stdexcept> 
 
using namespace std;
using namespace testsuitecontrol;
 
int main(int argc, char** argv)
{
int nRet = -1; 
try
{ 
initializeSession(); 
bool isRemoved =deleteFloatingTestsuite("dummy");
if (isRemoved)
{
cout<<"\"dummy\" testsuite was removed: succeed."<<endl;
}
else
{
cout<<"\"dummy\" testsuite was removed: failed."<<endl;
}
 
detachSession();
nRet = 0;
} 
catch(runtime_error& error)
{
cerr<<error.what()<<endl;
}
 
return nRet;

}
