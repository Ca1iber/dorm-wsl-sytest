Timing Wavetables - Example

WAVETBL gross_func_wtb

  PINS CP
         0        "d1:0 d2:0"
                 1        "d1:1 d2:0"
            brk      "" 

  PINS _MR
                 0        "d1:0 d2:1" 
                 1        "d1:1 d2:1"
            brk      "" 

  PINS mode ser_in
                 0        "d1:0" 
                 1        "d1:1" 
             brk      "" 

  PINS io_pins
                 0        "d1:0 r1:X" 
                 1        "d1:1 r1:X" 
                 2        "d1:Z r1:L" 
                 3        "d1:Z r1:H"
                 4        "d1:Z r1:X" 
             brk      "" 

  PINS ser_out
                 0        "r1:L" 
                 1        "r1:H" 
                 2        "r1:X"

WAVETBL spec_search_wtb

 
  PINS CP
                 0         "d1:0 d2:0"          0
                 1         "d1:1 d2:0"          1
               brk         "" 

  PINS _MR
                 0         "d1:0 d2:1"          0
                 1         "d1:1 d2:1"          1
               brk         ""

  PINS mode
                 0         "d1:1 d2:0 d3:1"     SBC_0
                 1         "d1:0 d2:1 d3:0"     SBC_1
                 2         "d1:0"               DNRZ_0
                 3         "d1:1"               DNRZ_1
               brk         "d1:0"

  PINS ser_in
                 0         "d1:1 d2:0 d3:1"     SBC_0
                 1         "d1:0 d2:1 d3:0"     SBC_1
               brk         ""

  PINS ser_out
                 0         "r1:L"               L
                 1         "r1:H"               H
                 2         "r1:X"               X

  PINS io_pins
                 0         "d1:1 d2:0 d3:1"     SBC_0
                 1         "d1:0 d2:1 d3:0"     SBC_1
                 2         "d1:1 r1:L"          L
                 3         "d1:1 r1:H"          H
                 4         "d1:1 r1:X"          X
