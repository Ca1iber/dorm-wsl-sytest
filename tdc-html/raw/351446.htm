Protocol management

Subtopics

  1. I2C protocol class template
  2. JTAG protocol class template
  3. SPI protocol class template
  4. Creating a protocol class

A protocol used for the RA library is a set of rules for how to do data transactions between devices and a tester. A protocol only defines a unique class.

The RA library natively supports protocol classes for I2C, JTAG, and SPI. The three protocol class templates are available. You can customize a template or create a protocol class based on the template and predefined transactions.

Available templates:
  • I2C protocol class template
  • JTAG protocol class template
  • SPI protocol class template
Predefined transactions:
  • I2C predefined transactions
  • JTAG predefined transactions
  • SPI predefined transactions

To use a protocol class, you must put the protocol .cpp file somewhere in your test method project. For example, /workspaces/david/TM/src/protocols/MyI2C.cpp, in which MyI2C.cpp is the protocol class file.

Example

Protocol for I2C with one class ProtocolI2C

class MyProtocolI2C : public ProtocolI2C {
public:
  MyProtocolI2C() {}
  virtual ~MyProtocolI2C() {}

  const static int SLAVE_ADDRESS = 0xE8;

  // Create singleton (Mandatory)
  RA_CREATE_SINGLETONE(MyProtocolI2C);


  virtual std::string getName() const { return "I2C"; }

  virtual double getClockSpeed() const { return 2.5 MHz ; }

  virtual void defineWriteRegister(I2C& i2c, ULONG64 address, ULONG64 data) {
    i2c.startCondition();
    i2c.sendByteWithAck(SLAVE_ADDRESS);
    i2c.sendByteWithAck(address / 256);
    i2c.sendByteWithAck(address % 256);
    i2c.sendByteWithAck(data, "inData");
    i2c.stopCondition();
  }
  virtual void defineReadRegister(I2C& i2c, ULONG64 address, int numberOfData) {
    i2c.startCondition();
    i2c.sendByteWithAck(SLAVE_ADDRESS);
    i2c.sendByteWithAck(address / 256);
    i2c.sendByteWithAck(address % 256);
    i2c.startCondition();
    i2c.sendByteWithAck( SLAVE_ADDRESS + 1);  ///< Read slave address

    if(numberOfData <= 1) {
      // One data
      i2c.receiveByteNoAck("outData");
    } else {
      // Burst read
      i2c.loopBegin(numberOfData - 1);
         i2c.receiveByteAckLow("outData");
      i2c.loopEnd();
      i2c.receiveByteNoAck("outData");
    }

    i2c.stopCondition();
  }
  virtual void defineCompareRegister(I2C& i2c, ULONG64 address, ULONG64 expectData, ULONG64 maskData) {
    i2c.startCondition();
    i2c.sendByteWithAck(SLAVE_ADDRESS);
    i2c.sendByteWithAck(address / 256);
    i2c.sendByteWithAck(address % 256);
    i2c.startCondition();
    i2c.sendByteWithAck(SLAVE_ADDRESS + 1);  ///< Read slave address
    i2c.compareByteNoAck(expectData, maskData);
    i2c.stopCondition();
  }

  virtual void composeRegisterWriteData(ParameterSetter& param, ULONG64 address, ULONG64 data) {
    param.setValue("inData", data);
  }
  virtual RA::ULONG64 composeRegisterReadData(ParameterGetter& param) {
    return param.getValue("outData");
  }

  /// I2C Pin Configuration
  virtual std::string getPinNameSCL() const { return "I2C_SCL"; }
  virtual std::string getPinNameSDA() const { return "I2C_SDA"; }

  /// Optional define write burst
  virtual void defineWriteArray(I2C& i2c, ULONG64 address, const vector<ULONG64>& arrayData) {
    i2c.startCondition();
    i2c.sendByteWithAck(SLAVE_ADDRESS);
    i2c.sendByteWithAck(address / 256);
    i2c.sendByteWithAck(address % 256);
    for(unsigned int i=0;i<arrayData.size();++i) {
      i2c.sendByteWithAck(arrayData.at(i));
    }
    i2c.stopCondition();
  }
};

REGISTER_PROTOCOL(MyProtocolI2C);
}
