DC event example: Setup per site

Example

DC Scale per site programming with SmarTest MAPI

The following example shows how to program an setup per site by DC events. The example can be copied into the run() method of a test method class.

Minimum one DC Scale pin that is named "DC_Scale_pin" and a pattern "Test_1_pattern" with about 100 dummy vectors are required. The example will run in offline mode.

virtual void run()
{
    double              commonValue;
    map <int,double>    valuePerSite;
    PATTERN_CONTROLLER  patternController;
    PATTERN             & pattern           = patternController.label("Test_1_pattern");
    static bool         firstRun            = true;
    EVENT_VF            event_VF            ("VF_runtime");
    
    ON_FIRST_INVOCATION_BEGIN();
    {
        //
        // Set the common value and the values per site
        //
        commonValue = 1.0 V;
    
        FOR_EACH_SITE_BEGIN();
        {
            valuePerSite[CURRENT_SITE_NUMBER()] = 0.7 + 0.1 V * CURRENT_SITE_NUMBER();
    
            cout << "Voltage for site " << CURRENT_SITE_NUMBER()
                 << " set to " << valuePerSite[CURRENT_SITE_NUMBER()] << " V" << endl;
        }
        FOR_EACH_SITE_END();
    
        //
        // First time setup
        //
        if (firstRun)
        {
            //
            // Set the common event data
            //
            event_VF.vForce(commonValue);
    
            cout << "First run: EVENT_VF defined" << endl;
    
            //
            // Insert the event into the pattern
            //
            pattern.vecEvents("DC_Scale_pin")
                    .clearAll   ()
                    .insert     (50, event_VF)
                    .setup      ();
    
            cout << "First run: EVENT_VF_1 inserted" << endl;
    
            firstRun = false;
    
        } // if (firstRun)
    
        //
        // Update to different values per site
        //
        event_VF.vForce(commonValue);
    
        pattern.vecEvents("DC_Scale_pin").update(event_VF);
    
        cout << "EVENT_VF set to the common value" << endl;
    
        FOR_EACH_SITE_BEGIN();
        {
            if (valuePerSite.find(CURRENT_SITE_NUMBER())!= valuePerSite.end())
            {
                event_VF.vForce(valuePerSite.at(CURRENT_SITE_NUMBER()));
    
                pattern.vecEvents("DC_Scale_pin").update(event_VF);
    
                cout << "Site " << CURRENT_SITE_NUMBER()
                     << " set to " << valuePerSite.at(CURRENT_SITE_NUMBER()) << " V" << endl;
            }
        }
        FOR_EACH_SITE_END();
    
        pattern.vecEvents("DC_Scale_pin").setup();
    }
    ON_FIRST_INVOCATION_END();

    return;
}

After the related test suite has been startet, you may inspect the modified pattern in the Pattern Editor. Follow the instructions in Configuring the layout of the editor if required to ensure that the Signal (TP) track of the "DC_Scale_pin" is shown.

Select vector 50 and check the settings of the DC event in the Pattern editor properties view. Use the Site Control Window to select the site in focus: Beginning with 800 mV on site 1, all sites should show different values for the Force Value of the DC event.
