Hidden upload examples for SmartRDI

Hidden upload using the SmartCalc (SMC) framework in means that SmartRDI performs the upload of the results in background threads.

Background threads using SmartCalc (SMC) in SmartRDI can be implemented as "thread per site" or as "common thread" (also known as "single thread"). The following examples use all the "thread per site" meaning that a separate thread is created for each active site.

For more information, refer to Thread per site in Using SmartCalc (SMC) with SmartRDI and to SmartCalc "Thread per site" and "Common thread".

Examples

Example 1: RF and DC (Thread per site)

virtual void run()
{
    RDI_INIT();

    ON_FIRST_INVOCATION_BEGIN();
    {
        rdi.hiddenUpload(TA::ALL);

        RDI_BEGIN();
        {
            rdi.rfm("id1").pin("A").signal(TA::MOD_OUT).freq(1e9).bw(2e6).execute();
            rdi.dc("id2").pin("A").iForce(1 mA).vMeas().execute();
            rdi.dc("id3").pin("B").vForce(1 V).iMeas().average(5).execute();
        }
        RDI_END();
    }
    ON_FIRST_INVOCATION_END();

    SMC_ARM();

    return;
}

public:

void SMC_backgroundProcessing()
{
    double pwr = rdi.id("id1").getPower();
    double r1 = rdi.id("id2").getValue();
    double r2 = rdi.id("id3").getValue();
}

Example 2: Digitizer and digital capture (Thread per site)

virtual void run()
{
    RDI_INIT();

    ON_FIRST_INVOCATION_BEGIN();
    {
        rdi.hiddenUpload(TA::ALL);

        RDI_BEGIN();
        {
            rdi.dgt("id1").pin("A").samples(1024).filter(1e6).FS(20e6).execute();
            rdi.digCap("id2").pin("A").samples(1024).execute();
        }
        RDI_END();
    }
    ON_FIRST_INVOCATION_END();

    SMC_ARM();

    return;
}

public:

void SMC_backgroundProcessing() 
{
    ARRAY_D wave1 = rdi.id("id1").getWaveform(); 
    ARRAY_I bits  = rdi.id("id2").getVectorBit(); 
}

Example 3: Error map (Thread per site)

virtual void run()
{
    RDI_INIT();

    ON_FIRST_INVOCATION_BEGIN();
    {
        rdi.hiddenUpload(TA::ALL); // Include the EMAP mode

        RDI_BEGIN();
        {
            rdi.smartVec("rb").pin("A").readBit(100).execute();
            rdi.emap("fc").label("bbb").pin("A").failCyc().execute();
        }
        RDI_END();
    }
    ON_FIRST_INVOCATION_END();

    SMC_ARM();

    return;
}

public:

void SMC_backgroundProcessing()
{
    rdi.id("rb").getReadBit();
    rdi.id("fc").getFailCyc();
}
