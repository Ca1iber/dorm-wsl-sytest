DIGITAL_WAVEFORM.checkEyeMask()

This function returns the eye mask test result plus the eye_height and the eye_width. The shift_mask_flag enables shifting the mask on the horizontal (timing) axis until a pass condition is found or the start position of the mask is reached again.

Syntax

void checkEyeMask(const STRING& pin, TM::SE_DIFF_MODE mode,
                  const ARRAY_I& maskxdata, const ARRAY_I& maskydata,
                  ARRAY_I& maskresult, DOUBLE& eye_height, DOUBLE& eye_width,
                  BOOLEAN shift_mask_flag) const
Parameters

pin

The pin to on which eye mask test is performed.

mode

The signal mode. It can be:

  • TM::SINGLE_ENDED: (for single-ended signals) The eye diagram voltage is referenced to the common(digital) ground.
  • TM::DIFF_SWING: (for differential signals) The eye diagram voltage is the voltage difference between the positive pin and negative pin (Vpos-Vneg).
  • TM::DIFF_COMMON: (for differential signals with common ground) The eye diagram voltage is (Vpos+Vneg)/2.

 

maskxdata

Mask data on x-axis

maskydata

Mask data on y-axis

maskresult

The eye mask test result. The elements of this array are 0 or 1, representing fail or pass.

eye_height

The height of the eye diagram.

eye_width

The width of the eye diagram.

shift_mask_flag

The flag to enable shifting the mask on the horizontal (timing) axis until a pass condition is found or the start position of mask is reached again.

Example

ARRAY_D pos_waveform_A;
ARRAY_D neg_waveform_A;
ARRAY_D dummy_A;
ARRAY_D ACCommonMode_waveform_A;
int stepsLowLevel;
int stepsHighLevel;
int bitStart;
int bitStop;
int maskXData[3];
int maskYData[3];
ARRAY_I maskResult;
DOUBLE  eyeHeigth;
DOUBLE  eyeWidth;
DOUBLE  TimingResolution;
STRING  pin(mposPins);
pin += ",";
pin += mnegPins;

stepsLowLevel  =  (int) (mLowLevel  / mLevelResolution);
stepsHighLevel =  (int) (mHighLevel / mLevelResolution);
TimingResolution = 400 ps / mSamplesPerUI ;
bitStart       = mXmode * mVectorStart;
bitStop        = mXmode * mVectorStop;
//configure the waveform capture for the pin
DIGITAL_WAVEFORM waveform_pos (pin);
waveform_pos.bitStart(bitStart);
waveform_pos.bitStop(bitStop);
waveform_pos.levelStart(stepsLowLevel);
waveform_pos.levelStop(stepsHighLevel);
waveform_pos.levelResolution(mLevelResolution);
waveform_pos.sampleStart(0);
waveform_pos.sampleStop(mSamplesPerUI-1);
waveform_pos.timingResolution(400 ps / mSamplesPerUI);

maskYData[0] = (int) (1.0 / mLevelResolution );
maskYData[1] = (int) (0.5 / mLevelResolution );
maskYData[2] = (int) (0.2 / mLevelResolution );


maskXData[0] = (int) (   0 ps / ( TimingResolution ) + 5 ) ;
maskXData[1] = (int) (   0 ps / ( TimingResolution ) + 5 ) ;
maskXData[2] = (int) (   0 ps / ( TimingResolution ) + 5 ) ;

cout << maskXData[0] * TimingResolution << " X-Data " << maskYData[0] * mLevelResolution << " Y-Data Point 0" << endl;
cout << maskXData[1] * TimingResolution << " X-Data " << maskYData[1] * mLevelResolution << " Y-Data Point 1" << endl;
cout << maskXData[2] * TimingResolution << " X-Data " << maskYData[2] * mLevelResolution << " Y-Data Point 2" << endl;
cout << TimingResolution << " Timing Resolution"  << endl;

waveform_pos.execute();
waveform_pos.checkEyeMask((STRING) mposPins, TM::SINGLE_ENDED, ARRAY_I( maskXData,sizeof(maskXData) / sizeof(int)),
                           ARRAY_I( maskYData,sizeof(maskYData) / sizeof(int)),maskResult, eyeHeigth,eyeWidth);
cout << "check eye mask test result for pin " << mposPins << "\n";
cout << "eye mask result = " << maskResult << "\n";
cout << "eye height      = " << eyeHeigth << "\n";
cout << "eye width       = " << eyeWidth << endl;
