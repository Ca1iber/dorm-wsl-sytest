Topic 124504

How to collect the hardware and software configuration with HVM API

About this task

HVM API provides the functionality to collect the software and hardware configuration of the test system. With HVM API you can collect the following configuration information:

Software

  • session ID of the online SmarTest session
  • profile information of a specified SmarTest session
  • version of the current SmarTest session
  • installed SmarTest versions
  • all IDs of the live SmarTest sessions
  • information of Linux operating system

Hardware

  • connected load board ID
  • connected external instrument information
  • connected probe card ID
  • ID list of all connected external instruments
  • test head type, DUT interface and list of DUT board ID
  • test board detail information like board type, slot number, module ID list, ...
  • test module information like module type, slot number, ...

The hardware configurations that can be collected depend on the SmarTest version. New boards are only supported in newer SmarTest versions.

For the execution of an HVM API program, refer to How to create and execute the application client with HVM API.

Before you begin

Make sure you are using SmarTest 6.2 or later. The SmarTest session is running online. Hardware information can only be collected when SmarTest is running in online mode.

Procedure

  1. Retrieve the software information with
    • GetOnlineSmartestSessionID
    • GetSmartestSessionProfile
    • ListInstalledSmartestVersions
    • ListSmartestSessionIDs
    • GetControllerInfo
  2. Retrieve the hardware information with
    • GetLoadboardInfo
    • GetInstrumentInfoById
    • GetProbeCardInfo
    • GetTesterInfo
    • GetTestHeadInfoByID
    • GetTestBoardInfoByID
    • GetTestModuleInfobyID
  3. Print out the hardware configuration of the test system.

Example

The following example retrieves the online SmarTest session ID. If an online session exists, it retrieves the session profile, lists all the installed SmarTest versions and retrieves the information of the operating system. Then the test head information is retrieved. All the detail information of test boards and test modules are retrieved and printed out.

using namespace xoc::hvmapi;
Status errObj;
ParamList qrResults, Results, params;
std::string testHeadID;
std::vector<std::string> instIdList;
std::vector<std::string> boardIdList;
std::vector<std::string> moduleIdList;
//----------------------Retrieve Software Information----------------
errObj = query("GetOnlineSmartestSessionID", qrResults);
if (errObj.isSuccess())
{
  session =  qrResults["SESSION_ID"];
}
else
{
  cout << "!!!Unable to collect tester information because there is no online Smartest Session" << endl;
  exit(0);
}
cout << endl;
errObj = query("GetSmartestSessionProfile", qrResults, ParamList(), session);
cout << "--------------------------------------------------------------------------" << endl;
cout << " Smartest Session Profile " << endl;
cout << "--------------------------------------------------------------------------" << endl;
if (errObj.isSuccess())
{
  cout<<"  VERSION " << qrResults["VERSION"] <<endl;
  cout<<"  ONLINE_MODE " << qrResults["ONLINE_MODE"] <<endl;
  cout<<"  USER_MODE " << qrResults["USER_MODE"] <<endl;
  cout<<"  LOGIN_USER " << qrResults["LOGIN_USER"] <<endl;
  cout<<"  ONLINE SESSION ID " << session <<endl;
}
else
{
  cout << "ERROR - Execution of GetSmartestSessionProfile Failed" << endl; 
}
cout << endl;
errObj = query("ListInstalledSmartestVersions", qrResults);
cout << "--------------------------------------------------------------------------" << endl;
cout << " Installed Versions of Smartest " << endl;
cout << "--------------------------------------------------------------------------" << endl;
if (errObj.isSuccess())
{
  cout<<"  CURRENT_VERSION " << qrResults["CURRENT_VERSION"] <<endl;
  vector<string> versionList = qrResults.valueArray("INSTALLED_VERSION_LIST");
  vector<string>::iterator it;
  int i = 1;
  cout<<"  VERSION_LIST:" << endl;
  for ( it=versionList.begin(); it!=versionList.end(); ++it, i++)
  {
    cout << "      version " << i << ": " << *it << endl;
  }
}
else
{
  cout << "ERROR - Execution of ListInstalledSmartestVersions Failed" << endl; 
}
cout << endl;
errObj = query("GetControllerInfo", qrResults);
cout << "--------------------------------------------------------------------------" << endl;
cout << " 93000 Tester Controller Information" << endl;
cout << "--------------------------------------------------------------------------" << endl;
if (errObj.isSuccess())
{
  cout << "OS Release:\t\t" << qrResults["OS_RELEASE"] << endl;
  cout << "OS Kernel:\t\t" << qrResults["OS_KERNEL"] << endl;
  cout << "Hostname:\t\t" << qrResults["HOSTNAME"] << endl;
}
else
{
  cout << "ERROR - Execution of GetControllerInfo Failed" << endl; 
}
cout << endl;  
//----------------------Retrieve Hardware Information----------------
errObj = query("GetTesterInfo", qrResults, params, session);
cout << "--------------------------------------------------------------------------" << endl;
cout << " 93000 Testhead Information" << endl;
cout << "--------------------------------------------------------------------------" << endl;
if (errObj.isSuccess())
{
  testHeadID = qrResults["TESTHEAD_ID"];
  cout << " Testhead ID:\t\t" << testHeadID << endl;
  instIdList = qrResults.valueArray("INSTRUMENT_ID_LIST");
  for(unsigned int i=0; i < instIdList.size(); i++)
  {
    cout << " Instrument ID:\t\t"<< instIdList[i] <<  endl;
  }
}
else
{
  cout << "ERROR - Execution of GetTesterInfo Failed" << endl; 
}
params["TESTHEAD_ID"] = testHeadID;
errObj = query("GetTestHeadInfoByID", qrResults, params, session);
if (errObj.isSuccess())
{
  cout << " Test Head Type: \t\t"<<qrResults["TESTHEAD_TYPE"]<<endl;
  cout << " DUT Interface: \t\t"<<qrResults["DUT_INTERFACE"]<<endl;
  boardIdList = qrResults.valueArray("BOARD_ID_LIST");
  for(unsigned int i=0; i < boardIdList.size(); i++)
  {
    cout << " Board ID:\t\t"<< boardIdList[i] <<  endl;
  }
}
else
{
  cout << "ERROR - Execution of GetTestHeadInfoByID Failed" << endl; 
}
cout << endl;  
cout << "--------------------------------------------------------------------------" << endl;
cout << " 93000 Instrument Information" << endl;
cout << "--------------------------------------------------------------------------" << endl;
for(unsigned int i=0; i < instIdList.size(); i++)
{
  params["INSTRUMENT_ID"] = instIdList[i];
  errObj = query("GetInstrumentInfoByID", qrResults, params, session);
  if (errObj.isSuccess())
  {
    cout <<" Instrument ID: "<<instIdList[i]<<"\tGPIB Address: "
           <<qrResults["GPIB_ADDRESS"]<<"\tDesc: "<<qrResults["DESCRIPTION"]<<endl;
  }
  else
  {
    cout <<" Instrument ID: ERROR cannot read"<<endl;
  }
}
cout << endl;
cout << "--------------------------------------------------------------------------" << endl;
cout << " 93000 Board Information" << endl;
cout << "--------------------------------------------------------------------------" << endl;
cout <<"\nBoard ID    Board Type   Slot   PN      SN   Vendor   Version"<<endl;
for(unsigned int i=0; i < boardIdList.size(); i++)
{
  params["BOARD_ID"] = boardIdList[i];
  errObj = query("GetTestBoardInfoByID", qrResults, params, session);
  if (errObj.isSuccess())
  {
    cout << boardIdList[i]<< "   ";
    cout <<qrResults["BOARD_TYPE"]<<"   ";
    cout <<qrResults["SLOT"]<<"   ";
    cout <<qrResults["PART_NUMBER"]<<"   ";
    cout <<qrResults["SERIAL_NUMBER"]<<"   ";
    cout <<qrResults["VENDOR"]<<"  ";
    cout <<qrResults["VERSION"]<<"   ";
    cout <<endl;
    if(qrResults["BOARD_TYPE"] == "RF_BOARD")
    {
      cout <<" HW Type: "    <<qrResults["HW_TYPE"]<< endl;
    }
    if(qrResults["BOARD_TYPE"] == "CHANNEL_BOARD")
    {
      cout <<" HW Type: "    <<qrResults["HW_TYPE"]<<endl;
      cout <<" Max Speed: "  <<qrResults["MAX_SPEED"]<<endl;
      moduleIdList = qrResults.valueArray("MODULE_ID_LIST");
      if(moduleIdList.size())
        cout <<"   Module ID   Module Type   Slot   PN      SN   Vendor   Version"<<endl;
      for(unsigned int j=0; j < moduleIdList.size(); j++)
      {
        params["MODULE_ID"] = moduleIdList[j];
        errObj = query("GetTestModuleInfoByID", qrResults, params, session);
        if (errObj.isSuccess())
        {
          cout <<"   "<<moduleIdList[j] << "   ";
          cout <<qrResults["MODULE_TYPE"]<<"   ";
          cout <<qrResults["SLOT"]<<"   ";
          cout <<qrResults["PART_NUMBER"]<<"   ";
          cout <<qrResults["SERIAL_NUMBER"]<<"   ";
          cout <<qrResults["VENDOR"]<<"   ";
          cout <<qrResults["VERSION"]<<"   ";
          cout <<endl;
        }
        else
        {
          cout <<" ERROR cannot read module info"<<endl;
        }
      }
      if(moduleIdList.size())
        cout <<"\nBoard ID    Board Type   Slot   PN      SN   Vendor   Version"<<endl;
    }
    if(qrResults["BOARD_TYPE"] == "DPS_BOARD")
    {
      cout <<" HW Type: "<<qrResults["HW_TYPE"]<<endl;
      moduleIdList = qrResults.valueArray("MODULE_ID_LIST");
      if(moduleIdList.size())
        cout <<"   Module ID   Module Type   Slot   PN      SN   Vendor   Version"<<endl;
      for(unsigned int j=0; j < moduleIdList.size(); j++)
      {
        params["MODULE_ID"] = moduleIdList[j];
        errObj = query("GetTestModuleInfoByID", qrResults, params, session);
        if (errObj.isSuccess())
        {
          cout <<"   "<<moduleIdList[j] << "   ";
          cout <<qrResults["MODULE_TYPE"]<<"   ";
          cout <<qrResults["SLOT"]<<"   ";
          cout <<qrResults["PART_NUMBER"]<<"   ";
          cout <<qrResults["SERIAL_NUMBER"]<<"   ";
          cout <<qrResults["VENDOR"]<<"   ";
          cout <<qrResults["VERSION"]<<"   ";
          cout <<endl;
        }
        else
        {
          cout <<" ERROR cannot read module info"<<endl;
        }
      }
      if(moduleIdList.size())
        cout <<"\nBoard ID    Board Type   Slot   PN      SN   Vendor   Version"<<endl;
    }
    if(qrResults["BOARD_TYPE"] == "ANALOG_BOARD")
    {
      cout <<" HW Type: "<<qrResults["HW_TYPE"]<<endl;
      cout <<" Number of Cores: "<<qrResults["NUMBER_OF_CORES"]<<endl;
      moduleIdList = qrResults.valueArray("MODULE_ID_LIST");
      if(moduleIdList.size())
        cout <<"   Module ID   Module Type   Slot   PN      SN   Vendor   Version"<<endl;
      for(unsigned int j=0; j < moduleIdList.size(); j++)
      {
        params["MODULE_ID"] = moduleIdList[j];
        errObj = query("GetTestModuleInfoByID", qrResults, params, session);
        if (errObj.isSuccess())
        {
          cout <<"   "<<moduleIdList[j] << "   ";
          cout <<qrResults["MODULE_TYPE"]<<"   ";
          cout <<qrResults["SLOT"]<<"   ";
          cout <<qrResults["PART_NUMBER"]<<"   ";
          cout <<qrResults["SERIAL_NUMBER"]<<"   ";
          cout <<qrResults["VENDOR"]<<"   ";
          cout <<qrResults["VERSION"]<<"   ";
          cout <<endl;
        }
        else
        {
          cout <<" ERROR cannot read module info"<<endl;
        }
      }
      if(moduleIdList.size())
        cout <<"   Module ID   Module Type   Slot   PN      SN   Vendor   Version"<<endl;
    }
  }
  else
  {
    cout <<" ERROR cannot read board info"<<endl;
  }
}
