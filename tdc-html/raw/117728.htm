DC event example: ADC linearity test

Setup test method

virtual void run()
{
  // Define the waveform
  ANALOG_WAVEFORM rampUp("rampUp");
  rampUp.definition(TM::RAMP)
        .direction(TM::POS)
        .samples(1024)
        .min(0.0).max(1.2);

  // Define the waveform force event and attach it to the pattern
  PATTERN_CONTROLLER pc;
  PATTERN & pattern(pc.label("ADC_lin_pat"));
  VEC_EVENTS & events(pattern.vecEvents("Analog_in"));

  EVENT_WAVEFORM_VF event_adc_ramp("event_adc_ramp");
  event_adc_ramp.vForce(rampUp);
  event_adc_ramp.sampleVectorDistance(32);

  // attach event to the pattern and download
  events.insert(85, event_adc_ramp);
  events.setup();
}

Execution test method

virtual void run()
{
  ARRAY_I    ADC_lin_capt, ADC_histogram;
  ARRAY_D    ile, dle;
  INT        num_miss_codes;
  INT        miss_codes[1024]; 
  INT        num_samples = 1024;
  LINEARtype lin_result;

  // Execute the test
  DIGITAL_CAPTURE_TEST( 1.0 );

  // Retrieve the capture values
  ADC_lin_capt = VECTOR( "VV_ADC_par" ).getVectors( );
  ADC_lin_capt.resize(num_samples);

  // Post-processing: Calculate the linearity errors
  DSP_HISTOGRAM(ADC_lin_capt, ADC_histogram, 0.5, 254.5, 256);
  DSP_ADC_HISTOGRAM(ADC_histogram, &lin_result, dle, ile,
                    &num_miss_codes,   miss_codes, MIN_RMS, 1.0);

  // Pass/Fail judgment
  TEST("Analog_in", "INL", lin_result.iamax, TM::CONTINUE);
  TEST("Analog_in", "DNL", lin_result.damax, TM::CONTINUE);
  TEST("Analog_in", "Missing_codes", num_miss_codes);   
}
