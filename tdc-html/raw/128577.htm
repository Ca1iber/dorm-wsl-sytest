Activation and deactivation of the offline result emulation

This section provides code examples on how to activate or deactivate the offline result emulation.

Activation

void run() {
  ON_FIRST_INVOCATION_BEGIN();

    OfflineResultEmulationConfig& oreConfig =
      OfflineResultEmulationConfig::instance();

    oreConfig.load("ore/oresetup.xml");
    oreConfig.enable();

  ON_FIRST_INVOCATION_END();
}

Automatic activation in offline mode

Example file Init_ORE.cpp:

#include "testmethod.hpp"

//for test method API interfaces
#include "mapi.hpp"
using namespace std;

/**
 * Test method class.
 *
 * For each testsuite using this test method, one object of this
 * class is created.
 */
class Init_ORE: public testmethod::TestMethod {
  string  mSetupFile;
  string  mEnableORE;

  void initialize()
  {
    addParameter("setupFile",
                 "string",
                 &mSetupFile,
                 testmethod::TM_PARAMETER_INPUT)
      .setComment("ORE setup file");

    addParameter("enableORE",
                 "string",
                 &mEnableORE,
                 testmethod::TM_PARAMETER_INPUT)
      .setDefault("0")
      .setOptions("0:1")
      .setComment("enable offline result emulation");
  }

  void run()
  {
    ON_FIRST_INVOCATION_BEGIN();
      int isOffline = 0;
      GET_SYSTEM_FLAG("offline", &isOffline);
      if (isOffline) {
        OfflineResultEmulationConfig& oreConfig =
          OfflineResultEmulationConfig::instance();
        if (mEnableORE != "0") {
          oreConfig.load(mSetupFile);
          oreConfig.enable();
        } else {
          oreConfig.disable();
          oreConfig.unload();
        }
      }
    ON_FIRST_INVOCATION_END();
  }

  void postParameterChange(const string& parameterIdentifier)
  {
    return;
  }

  const string getComment() const
  {
    string comment = " please add your comment for this method.";
    return comment;
  }
};
REGISTER_TESTMETHOD("Init_ORE", Init_ORE);

Deactivation

void run() {
  ON_FIRST_INVOCATION_BEGIN();

    OfflineResultEmulationConfig& oreConfig =
      OfflineResultEmulationConfig::instance();

    oreConfig.disable();
    oreConfig.unload();

  ON_FIRST_INVOCATION_END();
}

License overview

Offline result emulation (ORE) is available with the S&S_TOOLS_PACKAGE SW license. If this license is not available ORE is disabled.

For an overview of all licensed V93000 products by Product ID, as they are referenced on the Entitlement Certificates, see Licensed V93000 products.
