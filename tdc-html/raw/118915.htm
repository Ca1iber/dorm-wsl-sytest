Topic 118915

Hidden calculation

The difference between hidden calculation and parallel calculation is that the SMC_TEST() statement is moved into your SMC_backgroundProcessing() method. Once the SMC_backgroundProcessing() method has finished, the test results get buffered until a SMC_SYNCHRONIZE() is performed.

This way the entire calculation is hidden behind the 'other test(s)' as shown in the figure below. Since the test statement is within the SMC_backgroundProcessing() method, no return variables are required.

For delayed data logging, a new test method has to be generated, as shown in the example below.

Note If no limits are assigned to the test suite and TMLimit is not used, the limits assigned to SMC_SYNCHRONIZE() will be used.

Example

class sunSNRtestMethod: public testmethod::TestMethod
{
    private:

    ARRAY_D I_Capt[MS_MAX_SITES];

    protected:

    virtual void initialize()
    {
        addLimit("test_limit");
    }

    virtual void run()
    {
        ON_FIRST_INVOCATION_BEGIN();

            //
            // Setup and execute test as usual
            //
            DGT("pinDGT").enable();
            ...
            SMC_EXECUTE_TEST(1.0);

            LIMIT lim = GET_LIMIT_OBJECT("test_limit");

            //
            // Upload data per site and start calculation
            //
            FOR_EACH_SITE_BEGIN();

                INT site = CURRENT_SITE_NUMBER();

                //
                // Upload waveform
                //
                I_Capt[site] = DGT("pinDGT").getWaveform();

                //
                // Kick off calculation thread
                //
                SMC_ARM(lim);

            FOR_EACH_SITE_END();

        ON_FIRST_INVOCATION_END();

    } // virtual void run()

    public:

    void SMC_backgroundProcessing(LIMIT lim)
    {
        int     hardBin;
        int     TestNumber;
        string  softBin;
        DOUBLE  snr;
        DOUBLE  sPow;
        DOUBLE  nPow;

        int     site = SMC_SITE_NUMBER();

        //
        // Do calculation
        //
        DSP_SNR(I_Capt[site],&snr,&sPow,&nPow, 1024, 1, 3, 0.0, 0);

        //
        // Do judgment using the syntax
        //
        // SMC_TEST(const STRING& pin, const STRING& testname, const LIMIT& limit,
        //           DOUBLE value, Boolean Continue = FALSE, INT testNumber = -1,
        //           const STRING& softbin= "", INT hardbin = -1);
        ...
        ...

        //
        //Assign bin (note: selected HW and SW bin must exist in the flow) and testnumber
        //
        hardBin = 5; softBin = "SNRTest"; TestNumber = 1000;

        SMC_TEST("pinDGT","SNR", lim, snr,TM::CONTINUE, TestNumber,softBin,hardBin);

        hardBin = 7; softBin = "SigPowTest"; TestNumber = 2000;

        SMC_TEST("pinDGT","signalPow", lim, sPow,TM::CONTINUE, TestNumber,softBin,hardBin);

        hardBin = 8; softBin = "NoisePowTest"; TestNumber = 3000;

        SMC_TEST("pinDGT","noisePow", lim, nPow,TM::CONTINUE, TestNumber,softBin,hardBin);

    } // void SMC_backgroundProcessing(LIMIT lim)

}; // class sunSNRtestMethod: public testmethod::TestMethod

class runDelayedBinLog: public testmethod::TestMethod
{
    protected:

    virtual void run()
    {
        //
        // Wait until all calculations are done
        //
        SMC_SYNCHRONIZE();
        ...
        ...
    }

}; // class runDelayedBinLog: public testmethod::TestMethod
For handling of data within the SMC_backgroundProcessing() method, refer to Passing data to the background process in SmartCalc recommendations.
