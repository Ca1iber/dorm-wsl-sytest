MTP_MEMORY_PATTERN.getFailingCycleAddressList()

Retrieves the failing addresses for the specified data pin. The retrieved addresses are stored in an array with the order of failed cycles.

Therefore, you must specify or setFailAddressListOrdering(MTP::FAIL_CYCLE) before using this function.

Syntax

ARRAY_CYCLE_ADDRESSES* getFailingCycleAddressList (const int bit_index);
Parameter

Parameter

Description

bit_index

An integer which represents the bit whose failing address you want to retrieve.

0 represents the first data pin in the data pin list that is specified with setResultPins().

Return value

Returns an array which stores the retrieved addresses.

Example

void MemoryTest::run()
{
  MTP_MEMORY_TEST memoryTest;
  memoryTest.enableAddressBasedResults(true);
  memoryTest.setMaxFailingAddressesToReconstructPerPin(1024);
  memoryTest.setFailAddressListOrdering(MTP::ORDERING_CYCLE);

  // Execution is called only for first site
  ON_FIRST_INVOCATION_BEGIN();
  {
    memoryTest.execute();
  }
  ON_FIRST_INVOCATION_END();

  MTP_MEMORY_PATTERN &march6n = memoryTest.getMemroyPort("A").getMemoryPattern("march6n");
  ARRAY_CYCLE_ADDRESSES *pAddressList = march6n.getFailingCycleAddressList(0);
  LONG nr_addresses = pAddressList->size();

  // Loop over address list
  for(int i = 0 ; i < nr_addresses ; i++)
  {
    stringstream msg;
    msg << "cycle = " << (*pAddressList[i].cycle << endl;
    msg << "x = " << (*pAddressList)[i].x << endl;
    msg << "y = " << (*pAddressList)[i].y << endl;
    msg << "z = " << (*pAddressList)[i].z << endl;
    // Print out address to report window
    cout << msg.str();
    // Store address in datalog
    PUT_DATALOG(msg.str());
  }
}
