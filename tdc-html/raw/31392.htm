Topic 31392

Data display API use cases

Subtopics

  1. Data Plotting
  2. Cursor/Marker Analysis
  3. Customization

The API use cases are sample programs about Data Plotting.

A complete example is provided below to cover the whole process of creating plot, analyzing cursors/markers and customizing the plot.

Example

#include "math.h"
...
GraphManager * graphManager = GraphManager::getInstance();
graphManager->setTaskMode(Display::DISPLAY);
Graph graph;
Trace trace;
Cursor cursor;
Marker marker;
//Data plotting
//create a ramp waveform and display it in the sinal analyzer tool
vector<double> waveform1(256),waveform2(256);
for(int i=0;i<waveform1.size();i++)
{
waveform1[i]=i/2;
waveform2[i]=i/2*sin(2*M_PI*i/256);
}
 
trace.setTestLabel("Ramp").setWaveform(waveform1);
graph.setTrace(trace,1);
trace.setTestLabel("Sine").setWaveform(waveform2);
graph.setTrace(trace,2);
int id = graphManager->createGraph(graph);
 
//customization of the graph 
graph = graphManager->getGraph(id);
graph.setTitle("Sample");
graph.setBackGroundColor(Display::WHITE);
graph.getXAxis(1).setLabel("Sample X-axis label").setUnit("ms")
     .setMinRange(0).setMaxRange(400).setColor(Display::YELLOW);
graph.getYAxis(1).setVisible(0);
graphManager->setGraph(graph,id);
 
cursor=graph.getCursor();
cursor.setVisible(1);
cursor.setXMeasureColor(Display::BLACK);
cursor.setYMeasureColor(Display::GREEN);
cursor.setXReferenceColor(Display::RED);
cursor.setYReferenceColor(Display::BLUE);
graph.setCursor(cursor);
graphManager->setGraph(graph,id);
 
marker=graph.getMarker();
marker.setVisible(1);
marker.setUpMarkerColor(Display::RED);
marker.setDownMarkerColor(Display::BLUE);
graph.setMarker(marker);
graphManager->setGraph(graph,id);
graph.getTrace(1).setLineStyle(Display::SOLID)
     .setColor(Display::RED).setPointSymbol(Display::STAR);
graph.getTrace(2).setLineStyle(Display::DASH_DOT)
     .setColor(Display::YELLOW).setPointSymbol(Display::DIAMOND);
graphManager->setGraph(graph,id);
 
//double the sampling frequency
double frequency;
string unit;
graph.getTrace(1).getSamplingFrequency(&frequency,unit);
graph.getTrace(1).setSamplingFrequency(frequency*2,unit);
graph.setComment("Sample condition");
graphManager->setGraph(graph,id);
 
graph.getXAxis(1).autoScale(); 
graphManager->setGraph(graph,id);
 
graph.getYAxis(1).setVisible(1).setColor(Display::GREEN);
Axis x_axis=graph.getXAxis(1);
Axis y_axis=graph.getYAxis(1);
x_axis.getGrid().setStepSize(10).setColor(Display::GREEN);
 
graphManager->setGraph(graph,id);
