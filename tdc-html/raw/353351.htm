Creating a SmartProd client

SmartProd provides some APIs specially designed for monitoring switch_version events. You can use these APIs to create your own C++ program to monitor switch_version events.

About this task

Use SmartProd APIs to create a C++ program for monitoring switch_version operations.

Before you begin

Make sure that you have installed a SmartProd package on your workstation.

Procedure

  1. Add #include <smartMonitorApi.hpp> and using namespace smartmonitorapi; to the header part of your program.
  2. Create a monitor class with SmartMonitor, for example, class versionMonitor : public SmartMonitor.
  3. Consume specific events, for example, SmartMonitor::consume().
  4. Retrieve event information to get notified about version switch.
  5. Start your monitor instance with SmartMonitor::start().
  6. Stop your monitor instance with SmartMonitor::stop().

Results

The monitor client program is ready to be executed.

Example

Code example of a SmartProd client to monitor switch_version events

#include <iostream>
#include <unistd.h>
#include <SmartMonitorApi.hpp>
 
using namespace smartmonitorapi;
using namespace std;
 
class versionMonitor : public SmartMonitor
{
public:
  void consume(const Event& event){
    cout << "Receive Event:" << endl;
    cout << "EVENT TYPE : " << event.mEventType << endl;
    for(auto property : event.mEventProperties){
      cout << property.name << " : " << property.value[0] << endl;
    }
    cout << "Add your process here..." << endl;
    cout << endl;
  }
};
 
int main() 
{
 
    cout << "testing start for getting smartest version..." << endl;
    // test get current smartest version
    TestCell &testCell = TestCell::getInstance();
 
    SoftwareVersion ver = testCell.getCurrentTesterVersion();
    cout << "Current SmarTest Version string: " << ver.version << endl;
    cout << "Current SmarTest Version isDefault: " << ver.isDefault  << endl;
    cout << "Current SmarTest Version installationPath: " << ver.installationPath  << endl;
    cout << endl;
 
    // test get all installed smartest version list
    vector<SoftwareVersion> verList = testCell.getTesterVersions();
    cout << "Installed SmarTest Version: " << "(" << verList.size() << ")" << endl;
    for(auto ver: verList) {
        cout << "version: " << ver.version << endl;
        cout << "isDefault: " << ver.isDefault << endl;
        cout << "installationPath: " << ver.installationPath << endl;
        cout << endl;
        cout << endl;
    }
    cout << "testing end." << endl;
 
   try {
    versionMonitor versionMonitor;
    cout << "testing start for monitor..." << endl;
    versionMonitor.start();
    cout << "testing end..." << endl;
    sleep(5);
 
    return 0;
   }
   catch (string msg) {
     cout << msg << endl;
     return 0;
   }
   catch(...) {
    cout << "catch exception" << endl;
    return 0;
   }
   
}
