# Appendix C: Example Application Model

This application model demonstrates the sample code used for an automatically performed Wafer test using the GenericProber driver, and contains a number of application model variables which will be recognized by the STDF generation tool. These values are included in the STDF data generated by this tool.

```
hp93000,applicmodel,0.1
--      ---------------------------------------------------------------------
--
--      Example application model file for an automatically performed
--      WAFER test, using the GenericProber driver
--      ---------------------------------------------------------------------
--      DESCRIPTION:
--      First, the operator must enter the DIALOG_INPUT items:
--      Operator_id ....: string
--      Press the START button once a wafer prober has been connected
--      and all equipment is prepared for the test.
--      If the pause button is pressed the test pauses at DEVICE level
--      where a PAUSE dialog menu is displayed and the operator is asked to
--      enter the reason for the interruption.
--      The operator may change the prober driver's configuration or change
--      the driver's mode of operation during PAUSE.
--      ---------------------------------------------------------------------

START_DIALOG
{ testprogram:
    SCREEN_DIALOG
        * Operator_id   = DIALOG_INPUT           / REGEXP(string);
    ACTIONS
      OPER_NAM = CONST_INPUT({Operator_id});
      -- NOTE: The user does not need to make a special call
      -- in the App Model to invoke the STDF formatter.
      * HOST = CONST_INPUT(host);
    { lot:
        ACTIONS
          -- MIR field values
          STAT_NUM = CONST_INPUT(25);
          MODE_COD = CONST_INPUT(Q);
          RTST_COD = CONST_INPUT(N);
          PROT_COD = CONST_INPUT(B);
          BURN_TIM = CONST_INPUT(10);
          CMOD_COD = CONST_INPUT(A);
          LOT_ID  = CONST_INPUT(5);
          PART_TYP = CONST_INPUT (part 123);
          NODE_NAM = CONST_INPUT(node abc);
          TSTR_TYP = CONST_INPUT(93k SOC Tester);
          JOB_NAM = CONST_INPUT(Darwin);
          JOB_REV = CONST_INPUT(v1.5);
          SBLOT_ID = CONST_INPUT(2);
          OPER_NAM = CONST_INPUT(Joe Miller);
          TEST_COD = CONST_INPUT(FRN);
          TEST_TEMP = CONST_INPUT(40);
          USER_TXT = CONST_INPUT(wafer test);
          AUX_FILE = CONST_INPUT(additionalData);
          PKG_TYP  = CONST_INPUT(pkgA);
          FAMLY_ID = CONST_INPUT(Evolution Series);
          DATE_COD = CONST_INPUT(GET_TEST_INFO(date_time));
          FACIL_ID = CONST_INPUT(Houston);
          FLOOR_ID = CONST_INPUT(DL 416);
          PROC_ID = CONST_INPUT(fab process 175);
          OPER_FRQ = CONST_INPUT(normal);
          SPEC_NAM = CONST_INPUT(Darwin test spec);
          SPEC_VER = CONST_INPUT(v1.2);
          FLOW_ID = CONST_INPUT(Darwin-all);
          SETUP_ID = CONST_INPUT(59);
          DSGN_REV = CONST_INPUT(v1.5);
          ENG_ID = CONST_INPUT(158);
          ROM_COD = CONST_INPUT(375);
          SUPR_NAM = CONST_INPUT(Jane West);
          -- MRR field values
          DISP_COD = CONST_INPUT(2);
          EXC_DESC = CONST_INPUT(4);
          -- SDR field values (not all)
          HAND_TYP = CONST_INPUT(ITT Handler);
          HAND_ID = CONST_INPUT(59);
          CARD_TYP = CONST_INPUT(Initech probe card);
          CARD_ID = CONST_INPUT(43);
          LOAD_TYP = CONST_INPUT(Darwin load board);
          LOAD_ID = CONST_INPUT(27);
          CABL_TYP = CONST_INPUT(Interface cable XYZ);
          CABL_ID = CONST_INPUT(49);
        { cassette:
          LINE_DIALOG
            * cassette_id = DIALOG_INPUT           / REGEXP(string);
            { WAFER wafer:
                LINE_DIALOG
                 * wafer_id = DIALOG_INPUT         / REGEXP(string);
                ACTIONS
          -- WIR, WRR field values
          WAFER_ID = CONST_INPUT({PH_wafer_id});
          FABWF_ID = CONST_INPUT(fab wafer id 89);
          FRAME_ID = CONST_INPUT(wafer fram id 32);
          MASK_ID = CONST_INPUT(mask 145);
          STDF_FILE = CONST_INPUT(/tmp/{LOT_ID}.{wafer_id}.stdf);
                SUBLEVEL_COUNT wafer_log_count = CONST_INPUT(1);
                { wafer_log:
                    -- This level is used, to ensure that a wafer log file
                    -- is written after the DEVICE level is left.
                    -- The wafer_log level is executed exactly once for each
                    -- wafer.
                    { DEVICE [({WAFER_ID}),()] die:
                        {
                            DEVICE_TEST
                        }
                        ACTIONS
                        -- end of DEVICE level
                    }
                      -- end of wafer_log level
                }
                 -- end of wafer level
            }
             -- end of cassette level
        }
        -- end of lot level
    }
    -- end of test program
}
```
