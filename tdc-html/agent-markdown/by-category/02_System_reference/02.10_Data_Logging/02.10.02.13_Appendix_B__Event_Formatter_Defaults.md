# Appendix B: Event Formatter Defaults

This appendix contains the default Event Formatter settings file /opt/hp93000/soc/formatter/EventFormatter/.formatterrc.def

When you save your current settings for the Event Formatter, a copy of this file with the modified settings is written to your home directory with the name .formatterrc. Whenever you activate the Event Formatter again, this file is used instead of .formatterrc.def.

In this private .formatterrc file, you can change any setting according to your needs.

All possible settings and their effects are fully explained in the file itself:

```
######################################################################
#
# This file ($HOME/.formatterrc.def) is the configuration file for
# the Event Formatter.
#
# It can be edited to provide some user defined default settings for
# the Event Formatter.
#
######################################################################

# open_report_dialog_on_startup ==
#  0 : Report Dialog has to be opened manually.
#  1 : Report Dialog is automatically opened when the Event Formatter #      is started.

open_report_dialog_on_startup = 0;

# start_logging_on_startup_to ==
#  NULL : Do not log Event Formatter output automatically to a file when
#      the Event Formatter was started.
#  <filename> : Log Event Formatter output automatically to the specified
#      file when the Event Formatter was started. This happens the same way
#      as if you had selected "START Logging Formatter Output to file"
#      in the menu bar.
#      for example: start_logging_on_startup_to = "/tmp/EventFormatter.out"

start_logging_on_startup_to = NULL;

# auto_apply ==
#  0 : You have to press the 'Set Testsuite Flags' Button to apply the#      changes to the
#      test suite flags.
#  1 : All changes in Report Dialog are directly applied to the
#      test suite flags.

auto_apply = 0;

# output_all_testsuites ==
#  0 : Only test suites with test function events are printed to the
#      formatted output.
#  1 : All test suites (including bypassed test suites and test suites with
#      test functions that caused no data log information) are printed to
#      the formatted output.

output_all_testsuites = 0;

# output_width contains the number of columns that the width of the
# formatted output can have.  (output_width == 0) means that the width of
# the formatted output is not limited.

output_width = 0;

# ffv_spaces_between_pins contains the number of space characters between
# two pin names when printing first failing vectors in short mode
# (not in verbose mode).  (ffv_spaces_between_pins == 0) means that there
# is no free column between two pin names.

ffv_spaces_between_pins = 0;

# always_get_per_pin_pf_for_functional_tests ==
#  0 : The test suite flag "per pin on fail" is not automatically turned
#      on by the Event Formatter. If this flag is off, there is only a
#      pass/fail result for the whole test function available. This means
#      that the Event Formatter cannot distinguish between passed pins and
#      failed pins and therefore failed pins in ffv output cannot be
#      marked with a "#".
#  1 : The test suite flag "per pin on fail" is always turned on for
#      functional tests, so the test function returns a per pin pass/fail
#      result. This can increase the test function execution time.

always_get_per_pin_pf_for_functional_tests = 0;

# save_and_clear ==
#  0 : In the menu bar the entry for saving the data log stream is named
#      "Save Datalog Stream". When this entry is selected, the data log
#      stream is saved but not cleared. To clear the data log stream, you
#      have to select "Clear Datalog Stream" also.
#  1 : In the menu bar the entry for saving the data log stream is named
#      "Save and Clear Datalog Stream". When this entry is selected, the
#      data log stream is saved AND cleared.

save_and_clear = 1;

# shmoo_y_axis_direction ==
#  0 : (default) y axis is printed to down.
#  1 : y axis is printed to up.

shmoo_y_axis_direction = 0;

# shmoo_picture_generation ==
#  0 : (default) do not generate shmoo png picture
#  1 : generate shmoo png picture

shmoo_picture_generation  = 0;

# shmoo_set_name ==
#  Shmoo png naming convention
#  0 : (default) set shmoo png name by partId, site, mode and time stamp
#  For example:
#  If,
#    partId = P1.1
#    site = 1
#    mode = pass_fail
#  Then,
#    Shmoo Name = P1.1_Site_1_Mode_pass_fail_timestamp.png
#
#  1 : set shmoo png name by pattern/lable name, voltage Spec, site, mode and time stamp
#  For example:
#  If,
#    pattern = atpg_krait_lvmbist.bin
#    level specset = level
#    site = 2
#  Then,
#    Shmoo Name = atpg_krait_lvmbist_LevelSpecSet_level_Site_2_timestamp.png
#

shmoo_set_name = 0;

# shmoo_txt_generation ==
#  0 : do not generate shmoo txt (ASCII) file
#
#  1 : generate shmoo txt () ASCII file
#

shmoo_txt_generation   = 0;

# shmoo_pass_mark ==
#  . : pass
#  * : fail
#  Or other characters.
#  This option supports user-defined mark for show shmoo test results.
#

shmoo_pass_mark = '.';

# shmoo_fail_mark ==
#  . : pass
#  * : fail
#  Or other characters.
#  This option supports user-defined mark for show shmoo test results.
#

shmoo_fail_mark = '*';

# number_of_events_per_chunk and check_event_queue_period:
# To keep system load low, the event handler works in a loop: It sleeps
# for a defined time (= check_event_queue_period in ms), then it gets some
# (= number_of_events_per_chunk) events from the event queue, then it
# sleeps again and so on.

number_of_events_per_chunk = 500;
check_event_queue_period = 200;

# The following variables describe the initial checkbox settings for
# the report dialog.

report_dialog.functional.output_testsuites_that.pass = 1;
report_dialog.functional.output_testsuites_that.fail = 1;
report_dialog.functional.output_as_result.passed_pins = 0;
report_dialog.functional.output_as_result.failed_pins = 0;
report_dialog.functional.output_as_result.ffv_data = 0;

report_dialog.parametric.output_testsuites_that.pass = 1;
report_dialog.parametric.output_testsuites_that.fail = 1;
report_dialog.parametric.output_as_result.passed_pins = 0;
report_dialog.parametric.output_as_result.failed_pins = 0;
report_dialog.parametric.output_as_result.limits = 0;
report_dialog.parametric.output_as_result.shmoo_data = 0;

report_dialog.analog.output_testsuites_that.pass = 1;
report_dialog.analog.output_testsuites_that.fail = 1;
report_dialog.analog.output_as_result.passed_pins = 0;
report_dialog.analog.output_as_result.failed_pins = 0;
report_dialog.analog.output_as_result.limits = 0;
report_dialog.analog.output_as_result.waveforms = 0;

report_dialog.testmethod.output_testsuites_that.pass = 1;
report_dialog.testmethod.output_testsuites_that.fail = 1;
report_dialog.testmethod.output_as_result.passed_pins = 0;
report_dialog.testmethod.output_as_result.failed_pins = 0;
report_dialog.testmethod.output_as_result.limits = 0;
report_dialog.testmethod.output_as_result.ffv_data = 0;
report_dialog.testmethod.output_as_result.waveforms = 0;

report_dialog.user_procedures.output_user_procs = 0;
# device_cycle_name_width is the "Device Cycle Name" field width, when
# putting out FFV pin data.  Valid values are (6, 8, 10, 12, 14, 16)

device_cycle_name_width = 16;

report_dialog.ffv.verbose_mode = 0;
report_dialog.ffv.fine_result_granularity_mode = 0;
report_dialog.ffv.log_first_cycles = 1;
report_dialog.ffv.output_pins = All;
report_dialog.ffv.output_pins.use_format = 0;
report_dialog.ffv.output_pins.selected_format = NULL;
```
