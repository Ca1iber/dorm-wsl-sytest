The EVENT_WAVEFORM_VCVM (Voltage Clamp / Voltage Measurement) is a combined clamp voltage / voltage measure event class for defining events that generate waveforms in the clamp voltage mode.

This class is supported only for the [DC Scale XPS256 card](<../02.05_Test_Head_Cards/02.05.05.03_DC_Scale_XPS256_card.md>) in PMU operating mode.

## Class definition

The class derives from the [EVENT_BASE](<02.07.04.01.63.01_EVENT_BASE.md>) class.

The object that is created by this class has to be assigned to the [PATTERN_CONTROLLER](<02.07.04.01.142_PATTERN_CONTROLLER.md>) which is responsible for further handling.

This class is defined in the header file /opt/hp93000/soc/prod_com/include/MAPI/Events.h.

## Example

The following example creates two new DC events, fills the DC events with the data of two ANALOG_WAVEFORM objects and inserts the DC event into an already existing pattern. The sample vector distance is set to 100 for both waveforms with an offset of 50 vectors used for the second waveform. The programmed data are then read back from the event object and shown in the console window.

The pattern shows alternately DC events for the lower voltage clamp and for the upper voltage clamp in combination with a voltage measurement. Each of the events changes the lower clamp voltage or the upper clamp voltage but not both. The value of the currently not changed clamp voltage is shown as "0 V" but is grayed out and not applied to the hardware.

```
PATTERN_CONTROLLER patController;

ON_FIRST_INVOCATION_BEGIN();
{
    //
    // Create two waves for the lower and upper clamp voltage
    //
    ANALOG_WAVEFORM sineWave_low    ("sine_low");
    ANALOG_WAVEFORM sineWave_high   ("sine_high");
    INT             sineWaveSamples = 20;

    sineWave_low
        .definition (TM::SINE)
        .amplitude  (2.0 V)
        .offset     (-3.0 V)
        .samples    (sineWaveSamples)
        .periods    (2);

    sineWave_high
        .definition (TM::SINE)
        .amplitude  (4.0 V)
        .offset     (5.0 V)
        .samples    (sineWaveSamples)
        .periods    (2);

    //
    // Create two new event objects
    //
    EVENT_WAVEFORM_VCVM event_WAVEFORM_VCVM_low  ("WAVEFORM_VCVM_low");
    EVENT_WAVEFORM_VCVM event_WAVEFORM_VCVM_high ("WAVEFORM_VCVM_high");

    //
    // Set the event data
    //
    event_WAVEFORM_VCVM_low
        .vClampLow              (sineWave_low)
        .sampleVectorDistance   (100)
        .averages               (50)
        .outOfRangeCheck        (true);

    event_WAVEFORM_VCVM_high
        .vClampHigh             (sineWave_high)
        .sampleVectorDistance   (100)
        .averages               (40)
        .outOfRangeCheck        (false);

    //
    // Insert the events into a pattern
    //
    PATTERN & pattern = patController.label("...", "@");

    pattern.vecEvents("pin1").clearAll();

    pattern.vecEvents("pin1")
        .insert(100, event_WAVEFORM_VCVM_low)
        .insert(150, event_WAVEFORM_VCVM_high)
        .setup ();

    //
    // Read data from the event objects
    //
    ARRAY_D         readWave_low, readWave_high;
    INT             samples_1, distance_1, repeats_1, samples_2, distance_2, repeats_2;
    EVENT_VALUE_INT averages_1, averages_2;
    bool            outOfRangeCheck_1, outOfRangeCheck_2;

    readWave_low  = event_WAVEFORM_VCVM_low.getVClampLow();
    readWave_high = event_WAVEFORM_VCVM_high.getVClampHigh();

    cout << "wave size low  = " << readWave_low.size() << endl;
    cout << "wave size high = " << readWave_high.size() << endl;
    cout << endl;

    for (INT arrayIndex = 0; arrayIndex < readWave_low.size(); arrayIndex++)
    {
        cout << "readWave_low[" << arrayIndex << "] = " << readWave_low[arrayIndex] << endl;
    }

    cout << endl;

    for (INT arrayIndex = 0; arrayIndex < readWave_high.size(); arrayIndex++)
    {
        cout << "readWave_high[" << arrayIndex << "] = " << readWave_high[arrayIndex] << endl;
    }

    event_WAVEFORM_VCVM_low.getSamples                  (samples_1);
    event_WAVEFORM_VCVM_low.getSampleVectorDistance     (distance_1);
    event_WAVEFORM_VCVM_low.getRepeatCount              (repeats_1);
    event_WAVEFORM_VCVM_low.getAverages                 (averages_1);
    event_WAVEFORM_VCVM_low.getOutOfRangeCheck          (outOfRangeCheck_1);

    cout << endl;
    cout << "samples_1          = " << samples_1            << endl;
    cout << "distance_1         = " << distance_1           << endl;
    cout << "repeats_1          = " << repeats_1            << endl;
    cout << "averages_1         = " << averages_1           << endl;
    cout << "outOfRangeCheck_1  = " << outOfRangeCheck_1    << endl;

    event_WAVEFORM_VCVM_high.getSamples                 (samples_2);
    event_WAVEFORM_VCVM_high.getSampleVectorDistance    (distance_2);
    event_WAVEFORM_VCVM_high.getRepeatCount             (repeats_2);
    event_WAVEFORM_VCVM_high.getAverages                (averages_2);
    event_WAVEFORM_VCVM_high.getOutOfRangeCheck         (outOfRangeCheck_2);

    cout << endl;
    cout << "samples_2          = " << samples_2            << endl;
    cout << "distance_2         = " << distance_2           << endl;
    cout << "repeats_2          = " << repeats_2            << endl;
    cout << "averages_2         = " << averages_2           << endl;
    cout << "outOfRangeCheck_2  = " << outOfRangeCheck_2    << endl;
}
ON_FIRST_INVOCATION_END();
```
