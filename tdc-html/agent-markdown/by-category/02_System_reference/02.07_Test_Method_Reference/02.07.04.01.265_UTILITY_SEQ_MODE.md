## Example

```
UTILITY_SEQ_MODE utiSeqModeOn, utiSeqModeOff;

ON_FIRST_INVOCATION_BEGIN();

    //
    // Set utility block 1      in RW mode and
    // set utility block 2 to 5 in RD mode
    //
    utiSeqModeOn.pogoBlock(1).mode(TM::SEQ_POGO_MODE_RW );
    utiSeqModeOn.pogoBlock(2,5).mode(TM::SEQ_POGO_MODE_RD );
    utiSeqModeOn.execute();

    FUNCTIONAL_TEST();

    //
    // Disable the sequencer鈥慶ontrolled mode of all utility blocks
    //
    utiSeqModeOff.pogoBlock().mode(TM::SEQ_POGO_MODE_OFF);
    utiSeqModeOff.execute();

ON_FIRST_INVOCATION_END();
```
