The EVENT_VCVM (Voltage Clamp / Voltage Measurement) is a combined clamp voltage / voltage measure event class. This class provides the combined functionality of setting and measurement events in a single event.

This class is supported only for the [DC Scale XPS256 card](<../02.05_Test_Head_Cards/02.05.05.03_DC_Scale_XPS256_card.md>) in PMU operating mode.

## Class definition

The class derives from the [EVENT_BASE](<02.07.04.01.63.01_EVENT_BASE.md>) class.

The object that is created by this class has to be assigned to the [PATTERN_CONTROLLER](<02.07.04.01.142_PATTERN_CONTROLLER.md>) which is responsible for further handling.

This class is defined in the header file /opt/hp93000/soc/prod_com/include/MAPI/Events.h.

## Example

The following example creates three new DC events, sets the data of the DC events and inserts the DC events into an already existing pattern. The programmed data are then read back from the event objects and shown in the console window:

-

event_VCVM_1 uses member functions limitLow() and limitHigh() to define the limits. The lower clamp value is set by this DC event.

-

event_VCVM_2 uses the class LIMIT and the member function limits() to define the limits. The upper clamp value is set by this DC event.

-

event_VCVM_3 uses member functions limitLow() and limitHigh() to define the limits but the limits are cleared by member function clearLimits() to demonstrate how this function works. The lower clamp value is set by this DC event.

```
PATTERN_CONTROLLER patController;

ON_FIRST_INVOCATION_BEGIN();
{
    //
    // Create new event objects
    //
    EVENT_VCVM event_VCVM_1 ("VCVM_1");
    EVENT_VCVM event_VCVM_2 ("VCVM_2");
    EVENT_VCVM event_VCVM_3 ("VCVM_3");

    //
    // Set the event data
    //
    event_VCVM_1
        .vClampLow          (-5.0 V)    // vClampLow() OR vClampHigh()
        .averages           (50)
        .outOfRangeCheck    (true)
        .limitLow           (5.0 V)
        .limitHigh          (7.0 V);

    LIMIT event_VCVM_2_limits;

    event_VCVM_2_limits.low     (TM::GE, 8.0 V);
    event_VCVM_2_limits.high    (TM::LE, 9.0 V);
    event_VCVM_2_limits.unit    ("mA;-3;mA;-3");

    event_VCVM_2
        .vClampHigh         (22.0 V)    // vClampLow() OR vClampHigh()
        .averages           (50)
        .outOfRangeCheck    (true)
        .limits             (event_VCVM_2_limits);

    event_VCVM_3
        .vClampLow          (-2.0 V)    // vClampLow() OR vClampHigh()
        .limitLow           (10.0 V)    // Cleared with clearLimits()
        .limitHigh          (15.0 V)    // Cleared with clearLimits()
        .clearLimits        ();

    //
    // Insert the events into a pattern
    //
    PATTERN & pattern = patController.label("...", "@");

    pattern.vecEvents("pin1").clearAll();

    pattern.vecEvents("pin1")
        .insert(100, event_VCVM_1)
        .insert(150, event_VCVM_2)
        .insert(200, event_VCVM_3)
        .setup ();

    //
    // Read data from the event object event_VCVM_1
    //
    EVENT_VALUE_DOUBLE  lowerClamp_1, upperClamp_1, lowLimit_1, highLimit_1;
    EVENT_VALUE_INT     averages_1;
    bool                lowLimitDefined_1, highLimitDefined_1, outOfRangeCheck_1;

    event_VCVM_1.getVClampLow       (lowerClamp_1);
    event_VCVM_1.getVClampHigh      (upperClamp_1);
    event_VCVM_1.getAverages        (averages_1);
    event_VCVM_1.getOutOfRangeCheck (outOfRangeCheck_1);

    lowLimitDefined_1  = event_VCVM_1.getLimitLow   (lowLimit_1);
    highLimitDefined_1 = event_VCVM_1.getLimitHigh  (highLimit_1);

    cout << "lowerClamp_1          = " << lowerClamp_1                                      << endl;
    cout << "upperClamp_1          = " << upperClamp_1                                      << endl;
    cout << "averages_1            = " << averages_1                                        << endl;
    cout << "outOfRangeCheck_1     = " << outOfRangeCheck_1                                 << endl;
    cout << "lowLimit_1  (defined) = " << lowLimit_1  << " (" << lowLimitDefined_1  << ")"  << endl;
    cout << "highLimit_1 (defined) = " << highLimit_1 << " (" << highLimitDefined_1 << ")"  << endl;
    cout << endl;

    //
    // Read data from the event object event_VCVM_2
    //
    EVENT_VALUE_DOUBLE  lowerClamp_2, upperClamp_2, lowLimit_2, highLimit_2;
    bool                lowLimitDefined_2, highLimitDefined_2;

    event_VCVM_2.getVClampLow  (lowerClamp_2);
    event_VCVM_2.getVClampHigh (upperClamp_2);

    lowLimitDefined_2  = event_VCVM_2.getLimitLow   (lowLimit_2);
    highLimitDefined_2 = event_VCVM_2.getLimitHigh  (highLimit_2);

    cout << "lowerClamp_2          = " << lowerClamp_2                                      << endl;
    cout << "upperClamp_2          = " << upperClamp_2                                      << endl;
    cout << "lowLimit_2  (defined) = " << lowLimit_2  << " (" << lowLimitDefined_2  << ")"  << endl;
    cout << "highLimit_2 (defined) = " << highLimit_2 << " (" << highLimitDefined_2 << ")"  << endl;
    cout << endl;

    //
    // Read data from the event object event_VCVM_3
    //
    EVENT_VALUE_DOUBLE  lowerClamp_3, upperClamp_3, lowLimit_3, highLimit_3;
    bool                lowLimitDefined_3, highLimitDefined_3;

    event_VCVM_3.getVClampLow  (lowerClamp_3);
    event_VCVM_3.getVClampHigh (upperClamp_3);

    lowLimitDefined_3  = event_VCVM_3.getLimitLow   (lowLimit_3);
    highLimitDefined_3 = event_VCVM_3.getLimitHigh  (highLimit_3);

    cout << "lowerClamp_3          = " << lowerClamp_3                                      << endl;
    cout << "upperClamp_3          = " << upperClamp_3                                      << endl;
    cout << "lowLimit_3  (defined) = " << lowLimit_3  << " (" << lowLimitDefined_3  << ")"  << endl;
    cout << "highLimit_3 (defined) = " << highLimit_3 << " (" << highLimitDefined_3 << ")"  << endl;
}
ON_FIRST_INVOCATION_END();
```
