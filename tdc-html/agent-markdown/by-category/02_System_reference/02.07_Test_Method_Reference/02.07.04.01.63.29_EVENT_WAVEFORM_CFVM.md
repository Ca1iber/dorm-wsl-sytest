The EVENT_WAVEFORM_CFVM (Current Force / Voltage Measurement) is a combined force current / voltage measure event class for defining events that generate waveforms in the force current mode.

This class is supported only for the [DC Scale XPS256 card](<../02.05_Test_Head_Cards/02.05.05.03_DC_Scale_XPS256_card.md>) in PMU operating mode.

## Class definition

The class derives from the [EVENT_BASE](<02.07.04.01.63.01_EVENT_BASE.md>) class.

The object that is created by this class has to be assigned to the [PATTERN_CONTROLLER](<02.07.04.01.142_PATTERN_CONTROLLER.md>) which is responsible for further handling.

This class is defined in the header file /opt/hp93000/soc/prod_com/include/MAPI/Events.h.

## Example

The following example creates a new DC event, fills the DC event with the data of an ANALOG_WAVEFORM and inserts the DC event into an already existing pattern. The programmed data are then read back from the event object and shown in the console window.

```
PATTERN_CONTROLLER patController;

ON_FIRST_INVOCATION_BEGIN();
{
    //
    // Create a wave
    //
    ANALOG_WAVEFORM sineWave ("sine_1");

    sineWave
        .definition (TM::SINE)
        .amplitude  (20.0 mA)
        .offset     (5.0 mA)
        .samples    (20)
        .periods    (2);

    //
    // Create a new event object
    //
    EVENT_WAVEFORM_CFVM event_WAVEFORM_CFVM ("WAVEFORM_CFVM");

    //
    // Set the event data
    //
    event_WAVEFORM_CFVM
        .cForce                 (sineWave)
        .sampleVectorDistance   (50)
        .averages               (10)
        .outOfRangeCheck        (true)
        .clampChecks            (true, false);

    //
    // Insert the event into a pattern
    //
    PATTERN & pattern = patController.label("...", "@");

    pattern.vecEvents("pin1").clearAll();

    pattern.vecEvents("pin1")
        .insert(100, event_WAVEFORM_CFVM)
        .setup ();

    //
    // Read data from the event object
    //
    ARRAY_D         readWave;
    INT             samples, distance, repeats;
    EVENT_VALUE_INT averages;
    bool            outOfRangeCheck, clampCheckLow, clampCheckHigh;

    readWave = event_WAVEFORM_CFVM.getCForce();

    cout << "readWave size  = " << readWave.size() << endl;
    cout << endl;

    for (INT arrayIndex = 0; arrayIndex < readWave.size(); arrayIndex++)
    {
        cout << "readWave[" << arrayIndex << "] = " << readWave[arrayIndex] << endl;
    }

    event_WAVEFORM_CFVM.getSamples              (samples);
    event_WAVEFORM_CFVM.getSampleVectorDistance (distance);
    event_WAVEFORM_CFVM.getRepeatCount          (repeats);
    event_WAVEFORM_CFVM.getAverages             (averages);
    event_WAVEFORM_CFVM.getOutOfRangeCheck      (outOfRangeCheck);
    event_WAVEFORM_CFVM.getClampChecks          (clampCheckLow, clampCheckHigh);

    cout << endl;
    cout << "samples         = " << samples         << endl;
    cout << "distance        = " << distance        << endl;
    cout << "repeats         = " << repeats         << endl;
    cout << "averages        = " << averages        << endl;
    cout << "outOfRangeCheck = " << outOfRangeCheck << endl;
    cout << "clampCheckLow   = " << clampCheckLow   << endl;
    cout << "clampCheckHigh  = " << clampCheckHigh  << endl;
}
ON_FIRST_INVOCATION_END();
```
