```
   xyz_TASK task;

   ON_FIRST_INVOCATION_BEGIN();

 }

voidrun() {
xyz_TASK task1, task2;
   TASK_LIST tasks;

   ON_FIRST_INVOCATION_BEGIN();
     // setup
     tasks.add(task1);
     tasks.add(task2);
     tasks.execute();
   ON_FIRST_INVOCATION_END();

   DOUBLE val1 = task1.getValue("pin1");
   DOUBLE val2 = task2.getValue("pin1");
   // judge and log
 }
```

`getValue()` methods return hard-coded default values (for ex. 0.0)
