# Example codes of the most often used test control tasks

Some examples are provided below to show the usage of HVM API in test execution control.

## Query SmarTest session information

```
//query the session status of the current visible SmarTest sessions.
ParamList qrResults;
query("PingSmartestSessions", qrResults);
vector<string> idList = qrResults.valueArray("SESSION_ID_LIST");
vector<string> lifeStatusList = qrResults.valueArray("LIFE_STATUS_LIST");
for (unsigned int i=0; i< idList.size(); i++)
{
  string sid = idList[i];
  string lifeStatus = lifeStatusList[i];
  cout << "the session id "<< sid << " has the life status"
       << lifeStatus <<endl;
}
```

## Start a SmarTest session

```
//start an offline SmarTest session in supervisor mode.
xoc::hvmapi::ParamList results;

params["ONLINE_MODE"] = "OFFLINE";
params["USER_MODE"] = "SUPERVISOR";
errObj = xoc::hvmapi::operation("StartSmartestSession",results,params);
if (errObj.isSuccess())
{
  sSession=results["SESSION_ID"];
error_message =   errObj.code() + " : " +  errObj.text();
    sessionID = results["SESSION_ID"];

   }
  else
  {
      error_message = errObj.code() + " : " +  errObj.text();

   return 0;
}
```

## Change a device

```
//change the device to /home/demo/devices/74act299
Status errObj;
xoc::hvmapi::ParamList cmdParam;
session = ":0"; // Use the actual session ID
cmdParam["DEVICE_PATH"] = "/home/demo/devices";
cmdParam["DEVICE_NAME"] = "74act299";
cout<<"ChangeDevice started" <<endl;
errObj=xoc::hvmapi::command("ChangeDevice",cmdParam,session);
if(errObj.isSuccess())
{
  cout<<"ChangeDevice executed" <<endl;
}
else
{
  cout<<"ChangeDevice not executed" <<endl;
  cout<<"error code:" <<errObj.code() <<endl;
}
```

## Download a test program

```
//download the test program tp_tm to the test system
xoc::hvmapi::ParamList cmdParam;
cmdParam.setValue("TESTPROGRAM_NAME","tp_tm");

  cout <<"DownloadTestprogram executed" <<endl;
}
else
{
  cout <<"DownloadTestprogram not executed" <<endl;

```

## Query the test head system information

```
//get the online SmarTest ID
string sid;
ParamList qrResults,qrParams;
query("GetOnlineSmartestSessionID", qrResults);
sid = qrResults["SESSION_ID"];
//get the tester information
Status retval = query("GetTesterInfo", qrResults, ParamList(), sid);
if ( retval == SUCCESS )
{
  vector<string> iidList = qrResults.valueArray("INSTRUMENT_ID_LIST");
  printf("the TESTHEAD_ID is %s\n", qrResults["TESTHEAD_ID"]);//always 1
  foreach iid in iidList
  {
    printf("Instrument ID is %s\n", iid);
  }
}
//get the test head information
qrParams ["TESTHEAD_ID"] = qrResults["TESTHEAD_ID"];
query("GetTestHeadInfoByID", qrResults, qrParams, sid);
vector<string> bidList = qrResults.valueArray("BOARD_ID_LIST");
printf("The information for the testhead 1 is:\n");
printf("  <TESTHEAD_TYPE>: %s\n", qrResults ["TESTHEAD_TYPE"]);
printf("  < DUT_INTERFACE >: %s\n", qrResults ["DUT_INTERFACE"]);
printf("  <BOARD_ID_LIST>: \n"]);
foreach bid in bidList
{
    printf("     %s\n",  bid);
}
```

## Query the calibration status and execute DC update

```
//get the calibration status
string sid;
ParamList qrResults,qrParams;
query("GetOnlineSmartestSessionID", qrResults);
sid = qrResults["SESSION_ID"];
qrParams ["CAL_TYPE"] = "DC_UPDATE";
query("GetCalStatus", qrResults, qrParams, sid);

//execute the DC update calibration if it is outdate
ParamList opResults;
If(qrResults["CAL_DATA_VALIDITY"]=="EXPIRED")
{
  Status retVal = syncOperation("ExecDCUpdateCal", opResults,
                                ParamList(), sessionId);
  if ( ! retVal.isError() )
  {
    printf("successful execution, and execution result is %s\n",
          opResults["CAL_EXEC_STATUS"]);
  }
  else {
  printf("error (code=%s, description=%s), please check your operation",
      retVal.code(), retVal.text();
  }
}
```
