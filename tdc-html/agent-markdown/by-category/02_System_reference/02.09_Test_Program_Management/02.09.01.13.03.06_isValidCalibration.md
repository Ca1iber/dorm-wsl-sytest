# isValidCalibration()

This function returns whether a calibration is valid or not. It returns TRUE if the calibration is valid.

## Syntax

```
bool isValidCalibration (CalStatusType type,
            const char* cat,
            const char* pogoList);
```

  Parameters

 | Parameter  | Description |
 | type  | The calibration status type. It can be: |
 | CAL_TYPE_FULL  | Full calibration |
 | CAL_TYPE_UPDATE  | The incremental calibration of some specific steps |
 | cat  | The calibration category name from [calibrationCategoryList_next()](<02.09.01.13.03.01_calibrationCategoryList_next.md>). It can be: |
 | Core  | for the mandatory parts of a full calibration. |
 | ExtDc  | for the focus calibration. |
 | Fact  | for the one-time calibration done during the production of the hardware. |
 | Tmu  | for TMU calibration. It is optional and only needed when TMU is used. |
 | Update  | for the DC update calibration. The calibration has a shorter life time than the full calibration. Refer to the calibration periods for the individual cards in the [Specifications](<../02.02_Specifications/02.02_Specifications.md>). |
 | pogoList  | The list of cards or channels to be queried. It can be `NULL`, meaning all cards and channels.  |  Return value The queried state of the calibration as boolean value:

- TRUE: Calibration is valid
- FALSE: Calibration is invalid or has never been done

## Example

This is a sample program to do a calibration date query provided with SmarTest in the directory /opt/hp93000/soc/pws/calibration/example/.

For information on how to implement the example refer to [How to use the calibration API](<02.09.01.13.01_How_to_use_the_calibration_API.md>).

```
#include <stdio.h>

#include "/opt/hp93000/soc/pws/include/calibration.h"

void printDate (CalStatusEvent event,   // Calibration event: last or next calibration
                const char* name,       // Calibration category name
                const struct tm t)      // Time structure
{
    printf("Date of %s %-7s calibration: %4d-%02d-%02d %02d:%02d\n",
           event == CAL_EVENT_LAST ? "last" : "next",
           name, t.tm_year+1900, t.tm_mon+1, t.tm_mday, t.tm_hour, t.tm_min);

} // void printDate (CalStatusEvent event, const char* name, const struct tm t)

//
// Show date & time of calibration for all calibration categories
//
void showAllCalibrationDates (CalStatusEvent event) /* calibrationevent: last or next */
{
    //
    // Create a category iterator
    //
    CalibrationCategoryList* catList = calibrationCategoryList_new (CompleteCalibration, NULL);

    const char* catName = NULL;

    //
    // Loop through all category names
    //
    while (catName = calibrationCategoryList_next (catList), catName != NULL && *catName != '\0')
    {
        //
        // Get & print calibration date
        //
        printDate (event, catName, getCalibrationDate (event, CAL_TYPE_FULL, catName, NULL));
    }

    //
    // Delete iterator
    //
    calibrationCategoryList_delete (catList);

    //
    // Repeat the loop for update calibration
    //
    catList = calibrationCategoryList_new (UpdateCalibration, NULL);

    while (catName = calibrationCategoryList_next (catList), catName != NULL && *catName != '\0')
    {
        printDate (event, catName, getCalibrationDate (event, CAL_TYPE_UPDATE, catName, NULL));
    }

    calibrationCategoryList_delete (catList);

} // void showAllCalibrationDates (CalStatusEvent event)

//
// Print if calibration of all categories is valid or invalid
//
void isCalibrationValid()
{
    //
    // Create a category iterator
    //
    CalibrationCategoryList* catList = calibrationCategoryList_new (CompleteCalibration, NULL);

    const char* catName = NULL;

    //
    // Loop through all names of the category list
    //
    while (catName = calibrationCategoryList_next (catList), catName != NULL && *catName != '\0')
    {
        printf("%-7s calibration is %svalid\n", catName, isValidCalibration (CAL_TYPE_FULL, catName, NULL) ? "" : "in");
    }

    //
    // Delete iterator
    //
    calibrationCategoryList_delete (catList);

    //
    // Repeat the loop for update calibration
    //
    catList = calibrationCategoryList_new (UpdateCalibration, NULL);

    while (catName = calibrationCategoryList_next (catList), catName != NULL && *catName != '\0')
    {
        printf("%-7s calibration is %svalid\n", catName, isValidCalibration (CAL_TYPE_UPDATE, catName, NULL) ? "" : "in");
    }

    calibrationCategoryList_delete (catList);

} // void isCalibrationValid()

int main (void)
{
    printf("\nplease make sure that tester SW is running ..\n\n");

    //
    // Show calibration date of last full calibration
    //
    showAllCalibrationDates (CAL_EVENT_LAST);

    //
    // Show calibration date of next update calibration
    //
    showAllCalibrationDates (CAL_EVENT_NEXT);

    //
    // Is calibration valid ?
    //
    isCalibrationValid();

}
```
