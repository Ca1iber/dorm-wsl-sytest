# Example of analyzing the Iddq measurement results

This following example analyzes the test result of the test suite "prod_iddq" and print out the settings, result and result summary on the UI report window. You can use the Makefile of the DRLDump tool to compile this example.

```
#include <stdlib.h>
#include <unistd.h>
#include "/opt/hp93000/soc/com/include/drl.h"
#include "/opt/hp93000/soc/com/include/butil.h"
/* NOTE: dmalloc.h should always be the last file in the list of files to be included!*/
#ifdef USE_DMALLOC
  #include "dmalloc.h"
#endif
/*
 * The global variable <my_stream> holds the DRLDataStream from which
 * the events are read.
 */
DRLDataStream my_stream;
char indent_string[10];
int count=0;
double min_value, max_value, mean_value,sum_value;

/* Maps the failure code of the DRL to strings */
const char *MapFailureCode(int code)
{
  static char textual_failure_code[256];
  switch (code)
  {
    case DRLFailureCode_NA:
      return "NA";
    case DRLFailureCode_Pass:
      return "PASS";
    case DRLFailureCode_Fail:
      return "FAIL";
    case DRLFailureCode_Error:
      return "ERROR";
    case DRLFailureCode_Bypass:
      return "bypassed";
    default:
      sprintf(textual_failure_code,"*** Illegal failure code %d ***",code);
      return textual_failure_code;
  }
}

/* Maps the relation code of the DRL to strings */
const char *MapRelation(int relation)
{
  switch (relation)
  {
    case DRLCompareCode_NA:
      return "NA";
    case DRLCompareCode_Less:
      return "<";
    case DRLCompareCode_LessEqual:
      return "<=";
    default:
      return "*** ERROR";
  }
}
/* process the attributes of TestEventType
** print out the test name, test limit and per-pin result value
** and the min, max and mean value of the measurements.*/
void processTestEventAttribute(const DRLEventRec *event)
{
  int number_of_attributes, i,j;
  DRLAttributeRec attribute_rec;
 /* DRLAttributeRec attribute;*/

  number_of_attributes = DRLEventGetNoOfAttributes(event);
  fprintf(stderr,"%d ATTRIBUTS IN THE TEST EVENT\n",number_of_attributes);
  for (i=0; i<number_of_attributes; i++)
  {
    if (DRLEventGetAttribute(event,i,&attribute_rec))
    {
      fprintf(stderr,"Error while reading attributes");
      fprintf(stderr," - skipping event.\n");
      return;
    }
    switch(attribute_rec.type)
    {
      case TestNameAttrType:
        fprintf(stderr,"\nTestNameAttrType Received\n");
        fprintf(stderr,"%s%s\n",indent_string,attribute_rec.data.test_name_attr);
      break;
      case PassRangeAttrType:
        fprintf(stderr,"\nPassRangeAttrType Received\n");
        fprintf(stderr,"%s%.9f %s X %s %.9f [%s]\n",
        indent_string,attribute_rec.data.pass_range_attr->min_value,
        MapRelation(attribute_rec.data.pass_range_attr->min_value_relation),
        MapRelation(attribute_rec.data.pass_range_attr->value_max_relation),
        attribute_rec.data.pass_range_attr->max_value,
        attribute_rec.data.pass_range_attr->unit);
     break;
     case PinValueResultAttrType:
       fprintf(stderr,"\nPinValueResultAttrType Received\n");
       fprintf(stderr,"%s\"%s\"\n",indent_string,
       attribute_rec.data.pin_value_result_attr->description);
       for (j=0; j<attribute_rec.data.pin_value_result_attr->size; ++j)
       {

         if ( attribute_rec.data.pin_value_result_attr->pins[j].pin.name != NULL )
           fprintf(stderr,"%s%s", indent_string,
           attribute_rec.data.pin_value_result_attr->pins[j].pin.name);
         else
          fprintf(stderr,"%sno_pin_name_given", indent_string);
          fprintf(stderr,"%s(%d) = %f%s %s\n", indent_string,
          attribute_rec.data.pin_value_result_attr->pins[j].pin.index,
          attribute_rec.data.pin_value_result_attr->pins[j].value,
          attribute_rec.data.pin_value_result_attr->unit,
          MapFailureCode(attribute_rec.data.pin_value_result_attr->pins[j].failure_code));
        if(count==1)
          {
            min_value = attribute_rec.data.pin_value_result_attr->pins[j].value;
            max_value = attribute_rec.data.pin_value_result_attr->pins[j].value;
            mean_value = attribute_rec.data.pin_value_result_attr->pins[j].value;
            sum_value= attribute_rec.data.pin_value_result_attr->pins[j].value;
            fprintf(stderr,"min value of the test is:%f\n",min_value);
            fprintf(stderr,"max value of the test is:%f\n",max_value);
            fprintf(stderr,"mean value of the test is:%f\n",mean_value);
          }
          else
          {
            if(min_value > attribute_rec.data.pin_value_result_attr->pins[j].value)
            min_value = attribute_rec.data.pin_value_result_attr->pins[j].value;
            if(max_value < attribute_rec.data.pin_value_result_attr->pins[j].value)
            max_value = attribute_rec.data.pin_value_result_attr->pins[j].value;
            sum_value+=attribute_rec.data.pin_value_result_attr->pins[j].value;
            mean_value=sum_value/count;
            fprintf(stderr,"min value of the test is:%f\n",min_value);
            fprintf(stderr,"max value of the test is:%f\n",max_value);
            fprintf(stderr,"mean value of the test is:%f\n",mean_value);
          }
        }
      break;
    }
  }
}
/* process the attributes of TestSuiteEndEventType*/
void processTestSuiteEndAttribute(const DRLEventRec *event)
{
  int number_of_attributes, i;
  DRLAttributeRec attribute1;
  number_of_attributes = DRLEventGetNoOfAttributes(event);
  for (i=0; i<number_of_attributes; i++)
  {
    if (DRLEventGetAttribute(event,i,&attribute1))
    {
      fprintf(stderr,"Error while reading attributes");
      fprintf(stderr," - skipping event.\n");
      return;
    }
    switch(attribute1.type)
    {
      case SiteIdAttrType:
        fprintf(stderr,"Site %d: Test Suite prod_iddq Ended!\n",(int)attribute1.data.site_id_attr);
        fprintf(stderr,"%d tests done\n",count);
        break;
    }
  }

}

/* processes the TestSuiteStartEventType.*/
void ProcessTestSuiteStartEvent(const DRLEventRec *event)
{
  int number_of_attributes, i;
  DRLAttributeRec attribute_rec;
  char* mysuite_name;
  DRLEventRec event_rec;
  number_of_attributes = DRLEventGetNoOfAttributes(event);
  for (i=0; i<number_of_attributes; i++)
  {
    if (DRLEventGetAttribute(event,i,&attribute_rec))
    {
      fprintf(stderr,"Error while reading attributes");
      fprintf(stderr,"- skipping event.\n");
      return;
    }
    switch (attribute_rec.type)
    {
      case TestSuiteNameAttrType:
        mysuite_name = attribute_rec.data.test_suite_name_attr;
        break;
    }
  }
  if(!strcmp(mysuite_name, "prod_iddq"))
  {
    count++;
    fprintf(stderr,"\nTest suite %s Started!\n",mysuite_name);
    fprintf(stderr,"\n===============================\n");
    while(!DRLGetNextUnmaskedEvent(&my_stream, &event_rec))
    {
      switch(DRLEventGetType(&event_rec))
      {
        case TestEventType:
          fprintf(stderr,"\nTestEventType Received\n");
          processTestEventAttribute(&event_rec);
          fprintf(stderr,"\n-------------------------------\n");
        break;
        case TestSuiteEndEventType:
          processTestSuiteEndAttribute(&event_rec);
          return;
         break;
      }
    }
  }
}
/*Mask all event types except TestSuiteStart, TestEvent and TestSuiteEndEvent*/
void SetActiveEventMask(void)
{
  DRLEventType event_type;
  DRLEventMask my_event_mask;
  for (event_type=0; event_type<NumberOfEventTypes; event_type++)
  {
    my_event_mask.event_mask[event_type] = 1;
  }

  my_event_mask.size = NumberOfEventTypes;
  /** All events are masked and the event mask size is set correctly.*/

  my_event_mask.event_mask[TestSuiteStartEventType] = 0;
  my_event_mask.event_mask[TestEventType] = 0;
  my_event_mask.event_mask[TestSuiteEndEventType] = 0;
  DRLSetEventMask(&my_event_mask, &my_stream);
  /*
   * The TestSuiteStart, TestEvent and TestSuiteEndEvent are unmasked *
   *and the active event mask is set as specified in my_event_mask *
   */
}
/*******************************************************************
 *Waits for an event and branches to the appropriate event handler.
 * *****************************************************************/
void EventLoop(void)
{
  DRLEventRec event_rec;
  fprintf(stderr, "Press \'Ctrl-c\' to stop the Iddq result processor.\n");
  while (!DRLGetNextUnmaskedEvent(&my_stream, &event_rec))
  /*  Event Loop  */
  {   * Event Processing  */
    switch (DRLEventGetType(&event_rec))
    {
      case TestSuiteStartEventType:
        ProcessTestSuiteStartEvent(&event_rec);
      break;
     }
    /* Errors of DRLGetNextUnmaskedEvent are ignored */
  } /* End of Event Loop */
}

int main()
{
  int error_code;
  error_code = DRLStart(0,DRL_TRUNCATE | DRL_RESET_MASK | DRL_WAIT_FOR_EVENT,&my_stream);
  if (error_code!=0)
    fprintf(stderr,"DRLStart returned error code %d\n",error_code);
  else
  {
    SetActiveEventMask();
    EventLoop();
    DRLStop(&my_stream);
  }
  return 0;
}
```
