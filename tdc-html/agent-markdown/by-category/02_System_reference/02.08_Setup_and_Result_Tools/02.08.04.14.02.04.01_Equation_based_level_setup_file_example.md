# Equation based level setup file example

 Note Due to page width limitations, some of the following lines have been truncated.

```
hp93000,level,0.1
PSLV 1,0,0.01,HIZ,0,(vcc)

DRLV 1,300,3000,(aclkout,adata3,adata2,adata1,adata0)
RCLV 1,1550,1650,(aclkout)
RCLV 1,2500,3000,(adata3,adata2,adata1,adata0)
TERM 1,OFF,,,,,,(aclkout,adata3,adata2,adata1,adata0)
CLMP 1,OFF,-525,3725,(aclkout)
CLMP 1,OFF,-475,3775,(adata3,adata2,adata1,adata0)
LSUX 1
EQSP LEV,EQN,#9000001319EQNSET 1 "Equation set 1"

SPECS

vcc_level       [V]  # Base Vcc voltage
vcc_offset      [V]  # Offset to add to vcc_level
aclk_vil        [V]  # Initial VIL for aclk pins
aclk_vi_swing   [V]  # VIL-VIH swing for aclk pins
aclk_vo_swing   [V]  # VOL-VOH swing for aclk pins
adata_vil       [V]  # Initial VIL for adata pins
adata_vi_swing  [V]  # VIL-VIH swing for adata pins
adata_vo_swing  [V]  # VOL-VOH swing for adata pins

EQUATIONS

aclk_vih  = aclk_vil + aclk_vi_swing    # VIH for aclk
                                        # pins
aclk_vol  = aclk_vil + 1.5              # VOL for aclk
                                        # pins
aclk_voh  = aclk_vol + aclk_vo_swing    # VOH for aclk
                                        # pins
adata_vih = adata_vil + adata_vi_swing  # VIH for adata
                                        # pins
adata_vol = adata_vil + 1.5             # VOL for adata
                                        # pins
adata_voh = adata_vol + adata_vo_swing  # VOH for adata
                                        # pins

# derrived from dps set 1

DPSPINS vcc
vout   = vcc_level + vcc_offset
ilimit = 20
t_ms   = 0
offcurr= min

LEVELSET 1

PINS aclkout
vil= aclk_vil
vih= aclk_vih
vol= aclk_vol
voh= aclk_voh

PINS adata
vil= adata_vil
vih= adata_vih
vol= adata_vol
voh= adata_voh
@
EQSP LEV,SPS,#9000000751

EQNSET 1 "Equation set 1"

SPECSET 1 "example specification"

# SPECNAME       *****ACTUAL***** *****MINIMUM**** *****MAXIMUM**** UNITS COMMENT
vcc_level        0                                                  [  V]
vcc_offset       0.1              0                0.5              [  V]
aclk_vil         0                0              0.3                [  V]
aclk_vi_swing    4                3                4                [  V]
aclk_vo_swing    0.5              0.5              2                [  V]
adata_vil        0                0                0.9              [  V]
adata_vi_swing   3                0.5              4                [  V]
adata_vo_swing   0.5              0.1              2.5              [  V]
@
```

Example for equalization setup (droop compensation) in equation file

```
# Cross termination, differential comparitor
# set equalization to spec value (spec_eqi1...is defined in spec file)

LEVELSET 1
PINS pinIp pinIm
eqi1= spec_eqi1
eqi2= spec_eqi2
eqi3= spec_eqi3

PINS pinOp pinOm
term cross
eqo1= spec_eqo1
eqo2= spec_eqo2
```
