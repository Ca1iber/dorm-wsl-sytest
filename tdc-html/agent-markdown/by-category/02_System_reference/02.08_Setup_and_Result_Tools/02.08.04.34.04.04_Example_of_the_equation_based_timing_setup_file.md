# Example of the equation based timing setup file

#### Subtopics

- [Timing Wavetables - Example](<02.08.04.34.04.04.01_Timing_Wavetables_-_Example.md>)
- [Timing Equations - Example](<02.08.04.34.04.04.02_Timing_Equations_-_Example.md>)
- [Timing Specifications - Example](<02.08.04.34.04.04.03_Timing_Specifications_-_Example.md>)

hp93000,timing,0.1

```
PCLK 1,1,1000,(@)
CLKR 1,1000,10
BWDS 1,,,"",(CP,DS7,DS0,S1,S0,_MR,I/O7,I/O6,I/O5,I/O4,I/O3,I/O2,I/O1,I/O0)
ETDS 1,1,,"",(CP,DS7,DS0,S1,S0,_MR,Q7,Q0,I/O7,I/O6,I/O5,I/O4,I/O3,I/O2,I/O1,I/O0)
DCDT 1,""
TSUX 1,1
EQSP TIM,WVT,#9000001922
WAVETBL gross_func_wtb

  PINS CP
         0        "d1:0 d2:0"
                 1        "d1:1 d2:0"
            brk      ""

  PINS _MR
                 0        "d1:0 d2:1"
                 1        "d1:1 d2:1"
            brk      ""

  PINS mode ser_in
                 0        "d1:0"
                 1        "d1:1"
             brk      ""

  PINS io_pins
                 0        "d1:0 r1:X"
                 1        "d1:1 r1:X"
                 2        "d1:Z r1:L"
                 3        "d1:Z r1:H"
                 4        "d1:Z r1:X"
             brk      ""

  PINS ser_out
                 0        "r1:L"
                 1        "r1:H"
                 2        "r1:X"

#--------------------------------------------------------------------

WAVETBL spec_search_wtb

  PINS CP
                 0         "d1:0 d2:0"          0
                 1         "d1:1 d2:0"          1
               brk         ""

  PINS _MR
                 0         "d1:0 d2:1"          0
                 1         "d1:1 d2:1"          1
               brk         ""

  PINS mode
                 0         "d1:1 d2:0 d3:1"     SBC_0
                 1         "d1:0 d2:1 d3:0"     SBC_1
                 2         "d1:0"               DNRZ_0
                 3         "d1:1"               DNRZ_1
               brk         "d1:0"

PINS ser_in
                 0         "d1:1 d2:0 d3:1"     SBC_0
                 1         "d1:0 d2:1 d3:0"     SBC_1
               brk         ""

  PINS ser_out
                 0         "r1:L"               L
                 1         "r1:H"               H
                 2         "r1:X"               X

  PINS io_pins
                 0         "d1:1 d2:0 d3:1"     SBC_0
                 1         "d1:0 d2:1 d3:0"     SBC_1
                 2         "d1:1 r1:L"          L
                 3         "d1:1 r1:H"          H
                 4         "d1:1 r1:X"          X
@
EQSP TIM,EQN,#9000001444EQNSET 1 "gross_func_eqn"

SPECS
tpdIOx    [ns]

  TIMINGSET 1 "20MHz"  # Period and edge delays for a 20MHz test

    period = 50

    PINS CP
      d1 = 20
      d2 = 45

    PINS mode ser_in
      d1 = 0
      d2 = 30

    PINS _MR
      d1 = 0
      d2 = 15

    PINS ser_out
      r1 = 45

    PINS io_pins
      d1 = 2
      r1 = 45

# -----------------------------------------------------------

 EQNSET 2 "spec_search_eqn"

 SPECS
#-----------------------
#   Spec         Unit
# variables
#-----------------------

  f             [MHz]
  ts_Sx_CP      [ns]
  th_CP_Sx      [ns]
  ts_IOx_CP     [ns]
  th_CP_IOx     [ns]
  ts_DSx_CP     [ns]
  th_CP_DSx     [ns]
  tw_CP         [ns]
  tw_MR         [ns]
  trec_MR_CP    [ns]
  tpd_CP_Qx     [ns]
  tpd_CP_IOx    [ns]

 EQUATIONS
   p = 1/f*1000
   CP_ref = p/2
   ComplDelay = 2

 TIMINGSET 1 "std"

  period = p

    PINS CP
     d1 = CP_ref
     d2 = CP_ref + tw_CP

     PINS _MR
     d1 = CP_ref - trec_MR_CP - tw_MR
     d2 = CP_ref - trec_MR_CP

    PINS mode
     d1 = CP_ref - ts_Sx_CP - ComplDelay
     d2 = CP_ref - ts_Sx_CP
     d3 = CP_ref + th_CP_Sx

    PINS ser_in
     d1 = CP_ref - ts_DSx_CP - ComplDelay
     d2 = CP_ref - ts_DSx_CP
     d3 = CP_ref + th_CP_Sx

    PINS ser_out
     r1 = CP_ref + tpd_CP_Qx

    PINS io_pins
     d1 = CP_ref - ts_IOx_CP - ComplDelay
     d2 = CP_ref - ts_IOx_CP
     d3 = CP_ref + th_CP_IOx
     r1 = CP_ref + tpd_CP_IOx
@
EQSP TIM,SPS,#9000001339

EQNSET 1 "gross_func_eqn"

WAVETBL "gross_func_wtb"

CHECK all

SPECSET 1 "gross_func_specs"

# SPECNAME       *****ACTUAL***** *****MINIMUM**** *****MAXIMUM**** UNITS COMMENT
tpdIOx           0                                                  [ ns]

EQNSET 2 "spec_search_eqn"

WAVETBL "spec_search_wtb"

CHECK all

SPECSET 1 "spec_search_specs"

# SPECNAME       *****ACTUAL***** *****MINIMUM**** *****MAXIMUM**** UNITS COMMENT
f                50                                                 [MHz]
ts_Sx_CP         7                3                10               [ ns]
th_CP_Sx         1                -4               2                [ ns]
ts_IOx_CP        4                1.5              5                [ ns]
th_CP_IOx        1.2              -1               2                [ ns]
ts_DSx_CP        4.5              1.5              6                [ ns]
th_CP_DSx        1.5              -2               2                [ ns]
tw_CP            4                0                5                [ ns]
tw_MR            3.5              1                5                [ ns]
trec_MR_CP       3                0.5              5                [ ns]
tpd_CP_Qx        12               4                13.5             [ ns]
tpd_CP_IOx       12               4.5              15               [ ns]
@
NOOP "6.5.3",,,
```
