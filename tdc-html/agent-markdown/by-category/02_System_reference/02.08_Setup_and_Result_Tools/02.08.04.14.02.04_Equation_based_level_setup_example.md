# Equation based level setup example

#### Subtopics

- [Equation based level setup file example](<02.08.04.14.02.04.01_Equation_based_level_setup_file_example.md>)
- [Level specification: setup example](<02.08.04.14.02.04.02_Level_specification__setup_example.md>)

```
EQNSET 1 "Equation set 1"

SPECS

vcc_level       [V]  # Base Vcc voltage
vcc_offset      [V]  # Offset to add to vcc_level
aclk_vil        [V]  # Initial VIL for aclk pins
aclk_vi_swing   [V]  # VIL-VIH swing for aclk pins
aclk_vo_swing   [V]  # VOL-VOH swing for aclk pins
adata_vil       [V]  # Initial VIL for adata pins
adata_vi_swing  [V]  # VIL-VIH swing for adata pins
adata_vo_swing  [V]  # VOL-VOH swing for adata pins

EQUATIONS
aclk_vih  = aclk_vil + aclk_vi_swing    # VIH for aclk pins
aclk_vol  = aclk_vil + 1.5              # VOL for aclk pins
aclk_voh  = aclk_vol + aclk_vo_swing    # VOH for aclk pins

adata_vih = adata_vil + adata_vi_swing  # VIH for adata pins
adata_vol = adata_vil + 1.5             # VOL for adata pins
adata_voh = adata_vol + adata_vo_swing  # VOH for adata pins

MODULATION 1 "400Hz"
curve = SINE # only SINE is allowed
ac_level = 0.5 # [Vpp]
speed = 400 # [Hz]
duration = 20 # [ms]

MODULATION 2 "20kHz"
curve = SINE
ac_level = 0.5
speed = 20000
duration = 20

DPSPINS vcc
vout   = vcc_level + vcc_offset
ilimit = 20
t_ms   = 0
offcurr= min
ilimit = 900
t_ms = 0
offcurr = min
modulation = 1,2 # FW initial default is NONE.
mod_ilimit = 4000 # [mA]
mod_trig = ext # int|ext|imm

LEVELSET 1
PINS aclkout
vil= aclk_vil
vih= aclk_vih
vol= aclk_vol
voh= aclk_voh

PINS adata
vil= adata_vil
vih= adata_vih
vol= adata_vol
voh= adata_voh
```
