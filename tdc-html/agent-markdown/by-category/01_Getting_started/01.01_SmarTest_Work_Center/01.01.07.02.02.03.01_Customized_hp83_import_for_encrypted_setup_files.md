# Customized hp83_import for encrypted setup files

If you want to protect your setup files against unauthorized modification in a production environment, you should store them in encrypted form, see [How to crypt a file](<01.01.07.02.02.03.02_How_to_crypt_a_file.md>).

However, the encrypted files must be decrypted before they can be loaded into SmarTest. It is natural to confer this task to the hp83_import script. This script is called by the Test Program Explorer to load the setup files and can be customized by the user. For details, refer to [How to compile hp83_import](<01.01.07.02.02.03.03_How_to_compile_hp83_import.md>).

This section presents an example for such a customized import script, which assumes that the files have been encrypted with ccrypt. The example script has the following special features:

- It is not strictly speaking a script, but a compiled binary. The reason for this is that the encrypted setup files are associated with a pass word which hp83_import must know, but which should be hidden from the user.
- It ignores the [Setup Management Tool](<01.01.07.02.02.04_Setup_Management_Tool.md>). This seems to be justified by two reasons:

  - It is difficult to hide the decrypted subfiles from the user (see [Protection for your setup files](<01.01.07.02.02.03_Protection_for_your_setup_files.md>)).
  - The Setup Management tool is intended for distributed development, whereas the protection of the setup files is designed for a production environment.

- It decrypts the setup file only if

  - it has been called by SmarTest, and not by someone else, and
  - SmarTest has been called in Protected mode (-P).

## Example

The following code example, which assumes that the encryption pass word is "VERIGY93K", shows a possible implementation:

```
/*
 *-- system includes ------------------------------------------------------
 */
#include <stdio.h>
#include <stdlib.h>
#include <sys/types.h>
#include <unistd.h>
/*
 *-- module include -------------------------------------------------------
 */
#include "tpi_c.h"
#include "ci_types.h"
#include "libcicpi.h"
/*
 *-- defines -------------------------------------------------------------
 */
#define SA_PROCESS_INFO "/opt/hp93000/soc/pws/bin/saprocessinfo -sep ' '"
/*
 *-- globals variables ----------------------------------------------------
 */
/*
 *-- functions ------------------------------------------------------------
 */
int securityChecks()
{
    FILE *fd;
    char command[1024+1];
    char output[1024+1];

    fprintf(stderr, "\n\n");
    /**************************************************/
    /**          Are we online or offline?           **/
    /**************************************************/
    INT32 mode;
    GetOnlineMode(&mode);
    fprintf(stderr, "Online?\t\t\t%ld\t(0 = offline, 1 = online)\n", mode);
    /**************************************************/
    /**          Which user mode?                    **/
    /**************************************************/
    long usermode;
    GetUserMode(&usermode);
    fprintf(stderr, "Usermode?\t\t%ld\t(Protected mode has value 5)\n", usermode);
    if(5 != usermode) {
      return -1;
    }

    /**************************************************/
    /**          Who has called us?                  **/
    /**************************************************/
    fprintf(stderr, "The pid of our parent is %d\n", getppid());
    /**************************************************/
    /**Is our caller registered with the SmarTest SW?**/
    /**************************************************/
    sprintf(command,"%s| grep SA_PRC_TF | awk {'print $3'}", SA_PROCESS_INFO);
    fd=popen(command, "r");
    int registered_pid;

    fgets(output, 1024, fd);
    sscanf(output,"%d", &registered_pid);

    fprintf(stderr, "Pid registered for TF: %d\n", registered_pid);
    fclose(fd);

    if(registered_pid != getppid()) {
      return -1;
    }
    /**************************************************/
    /**        md5sum of our parent                  **/
    /**************************************************/
    sprintf(command, "md5sum /proc/%d/exe", getppid());
    fd=popen(command, "r");
    char temp[1024+1];
    fgets(output, 1024, fd);
    char md5sum[32+1];
    sscanf(output, "%s %s", md5sum, temp);
    fprintf(stderr, "md5sum of parent:\t%s\n", md5sum);
    fclose(fd);

    /**************************************************/
    /**            location of our parent            **/
    /**************************************************/
    sprintf(command, "/proc/%d/exe", getppid());
    int length=readlink(command, output, 1024);
    output[length]=0;
    fprintf(stderr, "loc of parent:\t\t%s\n", output);

    /**************************************************/
    /**   working directory of our parent            **/
    /**************************************************/
    sprintf(command, "/proc/%d/cwd", getppid());
    length=readlink(command, output, 1024);
    output[length]=0;
    fprintf(stderr, "cwd of parent:\t\t%s\n", output);
    /**************************************************/
    /**   root directory of our parent            **/
    /**************************************************/
    sprintf(command, "/proc/%d//", getppid());
    length=readlink(command, output, 1024);
    output[length]=0;
    fprintf(stderr, "root of parent:\t\t%s\n", output);
    return 0;
}
void decrypt(int argc, char* argv[])
{
  if(-1 == securityChecks()) {
    return;
  }
  char command[1024+1];
  if(0 == strcmp(argv[2], "config")) {
    fprintf(stderr, "Loading pins...\n");
    sprintf(command, "echo %s | ccrypt -c -k - %s", "VERIGY93K", argv[1]);
    system(command);
  }
  else if(0 == strcmp(argv[2], "level")) {
    fprintf(stderr, "Loading levels...\n");
    sprintf(command, "echo %s | ccrypt -c -k - %s", "VERIGY93K", argv[1]);
    system(command);
  }
  else if(0 == strcmp(argv[2], "timing")) {
    fprintf(stderr, "Loading timing...\n");
    sprintf(command, "echo %s | ccrypt -c -k - %s", "VERIGY93K", argv[1]);
    system(command);
  }
}
void plain(int argc, char* argv[])
{
  char command[1024+1];
  if(strstr(argv[1], ".gz")) {
    sprintf(command, "gunzip -c %s", argv[1]);
  }
  else if(strstr(argv[1], ".Z")) {
    sprintf(command, "uncompress -c %s", argv[1]);
  }
  else {
    sprintf(command, "cat %s", argv[1]);
  }
  system(command);
}
int main(int argc, char* argv[])
{
    HpInit();
    if(argc == 5) {
      if(0 == strcmp(argv[2], "config")) {
    decrypt(argc, argv);
      }
      else if(0 == strcmp(argv[2], "level")) {
    decrypt(argc, argv);
      }
      else if(0 == strcmp(argv[2], "timing")) {
    decrypt(argc, argv);
      }
      else {
    plain(argc, argv);
      }
    }
    exit(CI_CALL_PASS);                /* program completed without error */
    HpTerm();
}
```
