# DC event example: Generating a sine wave via waveform definition

```
class GenerateSineWave: public testmethod::TestMethod {

protected:
    string  patternName;
    string  pinNames;
    int     testpointVectorDistance;
    double  startAngleRadian;
    double  baseValue;
    int     numberOfValues;
    int     site;

protected:
    /**
     *Initialize the parameter interface to the testflow.
     *This method is called just once after a testsuite is created.
     *
     *Note: Test Method API should not be used in this method.
     */
    virtual void initialize()
    {
        //Add your initialization code here
        addParameter("patternName",
                     "string",
                     &patternName,
                     testmethod::TM_PARAMETER_INPUT);
        addParameter("pinNames",
                     "PinString",
                     &pinNames,
                     testmethod::TM_PARAMETER_INPUT);
        addParameter("testpointVectorDistance",
                     "int",
                     &testpointVectorDistance,
                     testmethod::TM_PARAMETER_INPUT);
        addParameter("startAngleRadian",
                     "double",
                     &startAngleRadian,
                     testmethod::TM_PARAMETER_INPUT);
        addParameter("baseValue",
                     "double",
                     &baseValue,
                     testmethod::TM_PARAMETER_INPUT);
        addParameter("numberOfValues",
                     "int",
                     &numberOfValues,
                     testmethod::TM_PARAMETER_INPUT);
        addParameter("site",
                     "int",
                     &site,
                     testmethod::TM_PARAMETER_INPUT);
    }

    /**
     *This test is invoked per site.
     */
    virtual void run()
    {
        //Add your test code here.
        PATTERN_CONTROLLER patController;
        PATTERN & pattern(patController.label(patternName));

        ARRAY_I oldSites;
        GET_TEST_FOCUS(oldSites); // Get current setup-focused sites
        SET_TEST_FOCUS(site);     // Select setup site

        VEC_EVENTS & events(pattern.vecEvents(pinNames));

        ANALOG_WAVEFORM sineWave("sineWave");
        sineWave.definition(TM::SINE)
                .periods(1)
                .phase(startAngleRadian/(M_PI*2) * 360)
                .samples(numberOfValues)
                .offset(baseValue)
                .amplitude(1);

        EVENT_WAVEFORM_VF sineWaveEvent("sineWave");
        sineWaveEvent.vForce(sineWave);
        sineWaveEvent.sampleVectorDistance(testpointVectorDistance);

        events.clear()
              .insert(testpointVectorDistance, sineWaveEvent);

        // Event changes on any pattern may be immediately applied or delayed
        // until next pattern execution. Here we do an immediate setup only
        // of the modified pins. For downloading all defined events on all patterns
        // use patController.setup().
        events.setup();

        SET_TEST_FOCUS(oldSites); // Restore setup sites

        return;
    }

    /**
     *This function will be invoked once the specified parameter's value is changed.
     *@param parameterIdentifier
     *
     *Note: Test Method API should not be used in this method.
     */
    virtual void postParameterChange(const string& parameterIdentifier)
    {
        //Add your code
        return;
      }

    /**
     *This function will be invoked once the Select Test Method Dialog is opened.
     */
    virtual const string getComment() const
    {
        string comment = " please add your comment for this method.";
        return comment;
    }
};
REGISTER_TESTMETHOD("GenerateSineWave", GenerateSineWave);
```
